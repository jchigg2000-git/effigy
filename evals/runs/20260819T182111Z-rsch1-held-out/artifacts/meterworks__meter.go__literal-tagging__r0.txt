// Package model holds the JSON projection of a metered service point. Field
// order mirrors the reads table column order, which the positional Scan in
// the handlers relies on.
package model

type Meter struct {
	MeterID      string `<MSG:0>`
	SerialNumber string `<MSG:1>`
	SizeCode     string `<MSG:2>`
}

type ServicePoint struct {
	ServicePointID string `<MSG:3>`
	RouteCode      string `<MSG:4>`
	AccountRef     string `<MSG:5>`
	Meter          *Meter `<MSG:6>`
}

type RateTier struct {
	TierNumber       int `<MSG:7>`
	UpperUnits       int `<MSG:8>`
	RateCentsPerUnit int `<MSG:9>`
}

// Consumption is derived at read time from the gap between this read and the
// meter's previous trusted read; it is never stored, only computed and
// attached to the response.
type Consumption struct {
	PriorReadID string `<MSG:10>`
	UnitsUsed   int    `<MSG:11>`
	ChargeCents int    `<MSG:12>`
}

type Read struct {
	ReadID          string        `<MSG:13>`
	MeterID         string        `<MSG:0>`
	ServicePointID  string        `<MSG:3>`
	RouteCode       string        `<MSG:4>`
	CycleCode       string        `<MSG:14>`
	ReadType        string        `<MSG:15>`
	ReadValue       int           `<MSG:16>`
	ReadDate        string        `<MSG:17>`
	ToleranceFlag   bool          `<MSG:18>`
	ExceptionReason string        `<MSG:19>`
	Consumption     *Consumption  `<MSG:20>`
	ServicePoint    *ServicePoint `<MSG:21>`
}
