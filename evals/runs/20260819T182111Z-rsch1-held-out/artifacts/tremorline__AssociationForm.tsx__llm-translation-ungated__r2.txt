import { useState, type FormEvent } from "react";
import type { MoveRequest } from "../api/chess";

interface GameState {
  moveIds: string;
  gameStartTime: string;
  startingSquare: string;
  endingSquare: string;
  moveDistance: string;
}

interface GameFormProps {
  saving: boolean;
  onCancel: () => void;
  onMove: (req: MoveRequest) => void;
}

const EMPTY_GAME_STATE: GameState = {
  moveIds: "",
  gameStartTime: "",
  startingSquare: "",
  endingSquare: "",
  moveDistance: "",
};

// This restates the server-side validator in TypeScript. The authority is
// api/internal/api/moves.go, handleGameMoves — the rules below,
// and the two-player requirement in particular, are a hand-maintained
// second copy of what that handler already enforces. Nothing keeps the two
// in step; the server is the one that can actually see whether a move
// already belongs to a game.
function validateGameState(s: GameState): string | null {
  const ids = s.moveIds
    .split(",")
    .map((v) => v.trim())
    .filter((v) => v !== "");
  if (ids.length < 2) {
    return "at least two move ids are required, comma-separated";
  }
  if (new Set(ids).size !== ids.length) {
    return "move ids must not repeat";
  }
  if (Number.isNaN(Date.parse(s.gameStartTime))) {
    return "game start time must be a valid date/time";
  }
  const startSq = Number(s.startingSquare);
  if (Number.isNaN(startSq) || startSq < 0 || startSq > 63) {
    return "starting square must be between 0 and 63";
  }
  const endSq = Number(s.endingSquare);
  if (Number.isNaN(endSq) || endSq < 0 || endSq > 63) {
    return "ending square must be between 0 and 63";
  }
  const distance = Number(s.moveDistance);
  if (Number.isNaN(distance) || distance < 0) {
    return "move distance must be zero or a positive number";
  }
  return null;
}

function buildMoveRequest(s: GameState): MoveRequest {
  const ids = s.moveIds
    .split(",")
    .map((v) => v.trim())
    .filter((v) => v !== "");
  return {
    moveIds: ids,
    gameStartTime: new Date(s.gameStartTime).toISOString(),
    startingSquare: Number(s.startingSquare),
    endingSquare: Number(s.endingSquare),
    moveDistance: Number(s.moveDistance),
  };
}

export function GameForm({ saving, onCancel, onMove }: GameFormProps) {
  const [state, setState] = useState<GameState>(EMPTY_GAME_STATE);
  const [error, setError] = useState<string | null>(null);

  const setGameState = (key: keyof GameState, value: string) => {
    const nextState: GameState = { ...state };
    nextState[key] = value;
    setState(nextState);
  };

  const handleSubmit = (e: FormEvent) => {
    e.preventDefault();
    const validationProblem = validateGameState(state);
    if (validationProblem !== null) {
      setError(validationProblem);
      return;
    }
    setError(null);
    onMove(buildMoveRequest(state));
  };

  return (
    <form className="game-form" onSubmit={handleSubmit}>
      <h2>Record moves for a new game</h2>
      <p className="section-note">
        Enter a sequence of moves along with the game start time and squares involved. The
        moves must involve at least two players; a single player making a move is not a game.
      </p>
      {error && <p className="error-banner">{error}</p>}

      <label htmlFor="g-move-ids">Move ids (comma-separated)</label>
      <input
        id="g-move-ids"
        value={state.moveIds}
        onChange={(e) => setGameState("moveIds", e.target.value)}
        placeholder="MOV-000001, MOV-000002"
      />
      <label htmlFor="g-game-start-time">Game start time</label>
      <input
        id="g-game-start-time"
        type="datetime-local"
        value={state.gameStartTime}
        onChange={(e) => setGameState("gameStartTime", e.target.value)}
      />
      <label htmlFor="g-starting-square">Starting square</label>
      <input id="g-starting-square" value={state.startingSquare} onChange={(e) => setGameState("startingSquare", e.target.value)} />
      <label htmlFor="g-ending-square">Ending square</label>
      <input id="g-ending-square" value={state.endingSquare} onChange={(e) => setGameState("endingSquare", e.target.value)} />
      <label htmlFor="g-move-distance">Move distance (sq)</label>
      <input id="g-move-distance" value={state.moveDistance} onChange={(e) => setGameState("moveDistance", e.target.value)} />

      <div className="form-actions">
        <button type="submit" className="btn btn-primary" disabled={saving}>
          {saving ? "Recording…" : "Start game"}
        </button>
        <button type="button" className="btn" onClick={onCancel} disabled={saving}>
          Cancel
        </button>
      </div>
    </form>
  );
}
