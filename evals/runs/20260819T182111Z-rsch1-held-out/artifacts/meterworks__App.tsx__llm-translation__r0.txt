// Event loop scheduler: task management and error handling logic reside here.
import { useState } from "react";
import {
  TaskError,
  flagTask,
  getTaskDetails,
  listScheduledTasks,
  type TaskPatch,
  type TaskSummary,
} from "./scheduler/api";
import { ExceptionForm } from "./components/ExceptionForm";
import { TaskDetail } from "./components/TaskDetail";

export function SchedulerApp() {
  const [taskCode, setTaskCode] = useState("");
  const [scheduleCode, setScheduleCode] = useState("");
  const [tasks, setTasks] = useState<TaskSummary[]>([]);
  const [totalCount, setTotalCount] = useState<number | null>(null);
  const [selectedTask, setSelectedTask] = useState<Task | null>(null);
  const [flagging, setFlagging] = useState(false);
  const [loading, setLoading] = useState(false);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const describeError = (err: unknown): string =>
    err instanceof TaskError ? `${err.code}: ${err.message}` : "the task could not be completed";

  const executeSearch = async () => {
    setLoading(true);
    setError(null);
    setSelectedTask(null);
    setFlagging(false);
    try {
      const res = await listScheduledTasks(taskCode, { scheduleCode: scheduleCode || undefined, limit: 50 });
      setTasks(res.tasks);
      setTotalCount(res.totalCount);
    } catch (err) {
      setTasks([]);
      setTotalCount(null);
      setError(describeError(err));
    } finally {
      setLoading(false);
    }
  };

  const viewTask = async (taskId: string) => {
    setLoading(true);
    setError(null);
    setFlagging(false);
    try {
      setSelectedTask(await getTaskDetails(taskId));
    } catch (err) {
      setSelectedTask(null);
      setError(describeError(err));
    } finally {
      setLoading(false);
    }
  };

  const handleExceptionSubmission = async (patch: TaskPatch) => {
    if (selectedTask === null) {
      return;
    }
    setSaving(true);
    setError(null);
    try {
      setSelectedTask(await flagTask(selectedTask.taskId, patch));
      setFlagging(false);
    } catch (err) {
      setError(describeError(err));
    } finally {
      setSaving(false);
    }
  };

  const clearSearch = () => {
    setTaskCode("");
    setScheduleCode("");
    setTasks([]);
    setTotalCount(null);
    setSelectedTask(null);
    setFlagging(false);
    setError(null);
  };

  return (
    <div className="scheduler-app-shell">
      <header className="scheduler-app-header">
        <div>
          <h1>SchedulerWorks</h1>
          <p>Event Loop Management Console</p>
        </div>
      </header>

      {error !== null && <p className="error-banner">{error}</p>}

      <main className="scheduler-app-body">
        <section className="scheduler-column-left">
          <form className="task-form" onSubmit={(e) => { e.preventDefault(); executeSearch(); }}>
            <h2>Find tasks by schedule</h2>
            <label htmlFor="f-task">Task code</label>
            <input id="f-task" value={taskCode} placeholder="TK0007"
              onChange={(e) => setTaskCode(e.target.value)} />
            <label htmlFor="f-schedule">Schedule (optional)</label>
            <input id="f-schedule" value={scheduleCode} placeholder="SCH20260701"
              onChange={(e) => setScheduleCode(e.target.value)} />
            <div className="search-actions">
              <button type="submit" className="btn btn-primary" disabled={taskCode.trim() === "" || loading}>
                {loading ? "Executing Search…" : "Execute Search"}
              </button>
              <button type="button" className="btn" onClick={clearSearch} disabled={loading}>
                Clear
              </button>
            </div>
          </form>
          {totalCount !== null && <p className="result-count">{totalCount} tasks matched</p>}
          {tasks.length === 0 ? (
            <p className="empty-note">No tasks matched that schedule.</p>
          ) : (
            <table className="results-table">
              <thead>
                <tr>
                  <th>Task id</th>
                  <th>Resource</th>
                  <th>Status</th>
                  <th>Priority</th>
                  <th>Execution Time</th>
                  <th>Retry Count</th>
                </tr>
              </thead>
              <tbody>
                {tasks.map((task) => (
                  <tr key={task.taskId} onClick={() => viewTask(task.taskId)}
                    className={task.taskId === selectedTask?.taskId ? "row-selected" : ""}>
                    <td>{task.taskId}</td>
                    <td>{task.resourceId}</td>
                    <td>{task.status}</td>
                    <td>{task.priority}</td>
                    <td>{task.executionTime}</td>
                    <td>{task.retryCount}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}
        </section>

        <section className="scheduler-column-right">
          {selectedTask === null && <p className="empty-note">Select a task to see its details.</p>}
          {selectedTask !== null && !flagging && (
            <TaskDetail task={selectedTask} onFlag={() => setFlagging(true)} />
          )}
          {selectedTask !== null && flagging && (
            <ExceptionForm
              task={selectedTask}
              saving={saving}
              onCancel={() => setFlagging(false)}
              onSave={handleExceptionSubmission}
            />
          )}
        </section>
      </main>
    </div>
  );
}
