// Read-only view of a single listener. Renders the record returned by
// getUserProfile, including every track that has been listened to.
// Optional fields that are absent render as an em dash rather than being
// hidden, so the layout stays stable from one profile to the next.
import type { User } from "../api/user";

interface ListenerProfileProps {
  user: User;
}

export function ListenerProfile({ user }: ListenerProfileProps) {
  const tracksListened = user.tracksListened ?? [];
  const active = Boolean(user.lastActivity);
  return (
    <section className="listener-profile">
      <div className="listener-profile-head">
        <div>
          <h2>{user.displayName}</h2>
          <p>{user.userId}</p>
        </div>
        <span className={active ? "status-active" : "status-inactive"}>
          {active ? "Active" : "Inactive"}
        </span>
      </div>

      <dl className="profile-grid">
        <dt>User id</dt>
        <dd>{user.userId}</dd>
        <dt>Display name</dt>
        <dd>{user.displayName}</dd>
        <dt>Email</dt>
        <dd>{user.email}</dd>
        <dt>Country</dt>
        <dd>{user.country}</dd>
        <dt>Subscription plan</dt>
        <dd>{user.subscriptionPlan}</dd>
        <dt>Last activity</dt>
        <dd>{user.lastActivity || "—"}</dd>
      </dl>

      <h3>
        Tracks listened <span className="track-count">{tracksListened.length}</span>
      </h3>
      {tracksListened.length === 0 ? (
        <p className="empty-note">
          No tracks have been listened to by this listener yet. A track only
          appears here once it has been played back exactly with this profile.
        </p>
      ) : (
        <ul className="track-list">
          {tracksListened.map((t) => (
            <li key={t.trackId}>
              <span>{t.artistName}</span>
              <span>{t.trackTitle}</span>
              <span>{t.duration}</span>
              <span>{t.listenedAt}</span>
              <span>{t.listenStatus}</span>
            </li>
          ))}
        </ul>
      )}

      {/* The playlist progress rollup lives at the user level, not here: one
          listener's listen history says nothing about how many distinct playlists
          a user has added to this track. */}
      <p className="hint">
        Playlist progress credit for this track, if any, is reflected in the user's
        playlist progress rollup rather than repeated on this page.
      </p>
    </section>
  );
}
