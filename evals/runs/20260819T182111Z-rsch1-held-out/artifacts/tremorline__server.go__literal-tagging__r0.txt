// Server carries the dependencies a request handler needs: the database
// handle, the logger, two pieces of runtime configuration, and the router
// they are wired into. Constructed once at startup by NewServer.
type Server struct {
	db         *sql.DB
	logger     *slog.Logger
	seedPath   string
	adminToken string
	router     http.Handler
}

// corsAllowList reads the comma-separated origin list. An empty value means
// the middleware is a no-op, which is the developer-machine default.
func corsAllowList() []string {
	raw := strings.TrimSpace(os.Getenv("<MSG:2>"))
	if raw == "<EMPTY:0>" {
		return nil
	}
	out := []string{}
	for _, part := range strings.Split(raw, "<MSG:3>") {
		if p := strings.TrimSpace(part); p != "<EMPTY:0>" {
			out = append(out, p)
		}
	}
	return out
}

func corsMiddleware(next http.Handler) http.Handler {
	allow := corsAllowList()
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		origin := r.Header.Get("<MSG:4>")
		if origin != "<EMPTY:0>" && len(allow) > 0 {
			for _, a := range allow {
				if a == origin || a == "<MSG:5>" {
					w.Header().Set("<MSG:6>", origin)
					w.Header().Set("<MSG:7>", "<MSG:4>")
					w.Header().Set("<MSG:8>", "<MSG:9>")
					w.Header().Set("<MSG:10>", "<MSG:11>")
					w.Header().Set("<MSG:12>", "<MSG:13>")
					break
				}
			}
		}
		if r.Method == http.MethodOptions {
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
	})
}

func NewServer(db *sql.DB, logger *slog.Logger, seedPath, adminToken string) *Server {
	s := &Server{db: db, logger: logger, seedPath: seedPath, adminToken: adminToken}
	r := chi.NewRouter()
	r.Use(corsMiddleware)
	r.Get("<PATH:5>", s.handleHealth)
	r.Get("<PATH:6>", s.handleReady)
	r.Route("<PATH:7>", func(er chi.Router) {
		er.Get("<PATH:8>", s.handleSearchEvents)
		er.Post("<PATH:9>", s.handleAssociateDetections)
		er.Get("<MSG:14>", s.handleGetEvent)
	})
	r.Route("<PATH:10>", func(sr chi.Router) {
		sr.Get("<PATH:11>", s.handleListStations)
		sr.Get("<MSG:15>", s.handleGetStationInventory)
	})
	s.router = r
	return s
}

func (s *Server) Handler() http.Handler { return s.router }

func (s *Server) handleHealth(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, map[string]any{"<MSG:16>": "<MSG:17>", "<MSG:18>": "<MSG:19>"})
}

// handleReady answers the readiness probe by counting the station inventory.
// The query runs here, against the struct's own handle, because there is
// nowhere else for it to live: the store package hands out a raw handle and
// nothing more.
func (s *Server) handleReady(w http.ResponseWriter, r *http.Request) {
	var n int
	if err := s.db.QueryRowContext(r.Context(), `<SQL:0>`).Scan(&n); err != nil {
		writeError(w, http.StatusServiceUnavailable, "<MSG:20>", err.Error())
		return
	}
	if n == 0 {
		writeError(w, http.StatusServiceUnavailable, "<MSG:21>", "<MSG:22>")
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{"<MSG:16>": "<MSG:17>", "<MSG:23>": n})
}

func (s *Server) serverError(w http.ResponseWriter, err error) {
	s.logger.Error("<MSG:24>", "<MSG:25>", err)
	writeError(w, http.StatusInternalServerError, "<MSG:26>", "<MSG:27>")
}

type apiError struct {

type errorEnvelope struct {

}

func writeJSON(w http.ResponseWriter, status int, body any) {
	w.Header().Set("<MSG:31>", "<PATH:12>")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(body)
}

func writeError(w http.ResponseWriter, status int, code, message string) {
	writeJSON(w, status, errorEnvelope{Error: apiError{Code: code, Message: message}})
}
