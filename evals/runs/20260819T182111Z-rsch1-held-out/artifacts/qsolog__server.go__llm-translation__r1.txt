package task_scheduler

import (
	"database/sql"
	"encoding/json"
	"log/slog"
	"net/http"
	"os"
	"strings"

	"github.com/go-chi/chi/v5"
)

// Scheduler carries the dependencies a task handler needs: the database
// handle, the logger, and the router they are wired into. Constructed once at
// startup by NewScheduler.
type Scheduler struct {
	db     *sql.DB
	logger *slog.Logger
	router http.Handler
}

// allowedOrigins reads the comma-separated origin list. An empty value means the
// middleware is a no-op, which is the developer-machine default.
func allowedOrigins() []string {
	raw := strings.TrimSpace(os.Getenv("TASK_SCHEDULER_ALLOWED_ORIGINS"))
	if raw == "" {
		return nil
	}
	out := []string{}
	for _, part := range strings.Split(raw, ",") {
		if p := strings.TrimSpace(part); p != "" {
			out = append(out, p)
		}
	}
	return out
}

func corsMiddleware(next http.Handler) http.Handler {
	allow := allowedOrigins()
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		origin := r.Header.Get("Origin")
		if origin != "" && len(allow) > 0 {
			for _, a := range allow {
				if a == origin || a == "*" {
					w.Header().Set("Access-Control-Allow-Origin", origin)
					w.Header().Set("Vary", "Origin")
					w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
					w.Header().Set("Access-Control-Allow-Headers", "Accept, Content-Type")
					w.Header().Set("Access-Control-Max-Age", "300")
					break
				}
			}
		}
		if r.Method == http.MethodOptions {
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
	})
}

// NewScheduler wires the task table. There is no admin or reload task: this
// service only ever reads what ingest already loaded at boot.
func NewScheduler(db *sql.DB, logger *slog.Logger) *Scheduler {
	s := &Scheduler{db: db, logger: logger}
	r := chi.NewRouter()
	r.Use(corsMiddleware)
	r.Get("/healthz", s.handleHealth)
	r.Get("/readyz", s.handleReady)
	r.Route("/tasks", func(tr chi.Router) {
		tr.Get("/search", s.handleSearchTasks)
		tr.Get("/{taskId}", s.handleGetTask)
	})
	r.Post("/logs", s.handleLogTask)
	r.Get("/progress", s.handleTaskProgress)
	r.Route("/workers", func(wr chi.Router) {
		wr.Get("/{workerId}", s.handleGetWorker)
		wr.Get("/{workerId}/summary", s.handleWorkerSummary)
	})
	s.router = r
	return s
}

func (s *Scheduler) Handler() http.Handler { return s.router }

func (s *Scheduler) handleHealth(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, map[string]any{"status": "ok", "service": "task_scheduler"})
}

// handleReady answers the readiness probe by counting logged tasks. The
// query runs here, against the struct's own handle, because there is nowhere
// else for it to live: the storage package hands out a raw handle and nothing
// more.
func (s *Scheduler) handleReady(w http.ResponseWriter, r *http.Request) {
	var n int
	if err := s.db.QueryRowContext(r.Context(), `SELECT count(*) FROM tasks`).Scan(&n); err != nil {
		writeError(w, http.StatusServiceUnavailable, "db_error", err.Error())
		return
	}
	if n == 0 {
		writeError(w, http.StatusServiceUnavailable, "not_seeded", "no tasks logged")
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{"status": "ok", "tasks": n})
}

func (s *Scheduler) serverError(w http.ResponseWriter, err error) {
	s.logger.Error("request failed", "err", err)
	writeError(w, http.StatusInternalServerError, "internal", "internal server error")
}

type taskError struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

type errorWrapper struct {
	Error taskError `json:"error"`
}

func writeJSON(w http.ResponseWriter, status int, body any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(body)
}

func writeError(w http.ResponseWriter, status int, code, message string) {
	writeJSON(w, status, errorWrapper{Error: taskError{Code: code, Message: message}})
}
