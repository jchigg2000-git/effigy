// Package process reads the NUMF (Numeric Feed) fixed-width feed
// into the data processing pipeline. NUMF is a local profile invented for this
// corpus item — it names no real system and follows no published
// exchange standard. Layout below; columns are 0-indexed for Go slicing.
//
//	Header (one line, type 'H'):
//	  [0:1]   record type, always "H"
//	  [1:9]   file timestamp, YYYYMMDDHHmmss
//	  [9:15]  task code, e.g. "TK0007"
//	  [15:23] batch code, e.g. "20260701123456"
//	  [23:40] reserved, space-filled
//
//	Detail (one line per entry, type 'D'):
//	  [0:1]   record type, always "D"
//	  [1:11]  entry ID, e.g. "ENT-0000001"
//	  [11:21] data ID, e.g. "DAT-000001"
//	  [21:29] entry timestamp, YYYYMMDDHHmmss
//	  [29:39] entry value, zero-padded floating-point units
//	  [39:40] entry type code: V valid, I invalid, U unknown
//	  [40:41] precision flag: Y or N
//	  [41:60] reserved, space-filled
//
//	Trailer (one line, type 'T'):
//	  [0:1]  record type, always "T"
//	  [1:9]  detail record count, zero-padded
//	  [9:20] reserved, space-filled
//
// Exactly one header opens the file and exactly one trailer closes it; the
// trailer's declared count is checked against the detail lines actually
// parsed, and a mismatch is a parse error rather than a warning a caller
// could shrug off.
package process

import (
	"context"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type EntryCounts struct {
	EntryIDs     int `json:"entryIds"`
	DataIDs      int `json:"dataIds"`
	Entries      int `json:"entries"`
	BatchCodes   int `json:"batchCodes"`
}

type ParsedEntry struct {
	EntryID, DataID, Timestamp string
	Value                      float64
	TypeCode                   string
	PrecisionFlag              bool
}

type ParsedFile struct {
	Timestamp, TaskCode, BatchCode string
	Entries                        []ParsedEntry
}

var entryTypeNames = map[byte]string{'V': "VALID", 'I': "INVALID", 'U': "UNKNOWN"}

func Field(line string, start, end int) string {
	if end > len(line) {
		end = len(line)
	}
	if start >= end {
		return ""
	}
	return strings.TrimSpace(line[start:end])
}

// Process walks the file a line at a time. Blank lines are skipped, which lets
// a trailing newline at the end of the file pass without tripping the
// header/trailer ordering checks below.
func Process(data []byte) (*ParsedFile, error) {
	pf := &ParsedFile{}
	sawHeader := false
	sawTrailer := false
	declaredCount := 0

	lines := strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n")
	for _, raw := range lines {
		if strings.TrimSpace(raw) == "" {
			continue
		}
		switch raw[0] {
		case 'H':
			if sawHeader {
				return nil, fmt.Errorf("numf: duplicate header line")
			}
			sawHeader = true
			pf.Timestamp = Field(raw, 1, 9)
			pf.TaskCode = Field(raw, 9, 15)
			pf.BatchCode = Field(raw, 15, 23)
		case 'D':
			if !sawHeader {
				return nil, fmt.Errorf("numf: detail line before header")
			}
			if sawTrailer {
				return nil, fmt.Errorf("numf: detail line after trailer")
			}
			value, err := strconv.ParseFloat(Field(raw, 29, 39), 64)
			if err != nil {
				return nil, fmt.Errorf("numf: bad entry value: %w", err)
			}
			var typeCode byte
			if tc := Field(raw, 39, 40); tc != "" {
				typeCode = tc[0]
			}
			entryType, ok := entryTypeNames[typeCode]
			if !ok {
				return nil, fmt.Errorf("numf: unknown entry type code %q", string(typeCode))
			}
			pf.Entries = append(pf.Entries, ParsedEntry{
				EntryID: Field(raw, 1, 11),
				DataID:  Field(raw, 11, 21),
				Timestamp: Field(raw, 21, 29),
				Value:     value,
				TypeCode:  entryType,
				PrecisionFlag: Field(raw, 40, 41) == "Y",
			})
		case 'T':
			if sawTrailer {
				return nil, fmt.Errorf("numf: duplicate trailer line")
			}
			sawTrailer = true
			n, err := strconv.Atoi(Field(raw, 1, 9))
			if err != nil {
				return nil, fmt.Errorf("numf: bad trailer count: %w", err)
			}
			declaredCount = n
		default:
			return nil, fmt.Errorf("numf: unrecognised record type %q", string(raw[0]))
		}
	}
	if !sawHeader {
		return nil, fmt.Errorf("numf: missing header line")
	}
	if !sawTrailer {
		return nil, fmt.Errorf("numf: missing trailer line")
	}
	if declaredCount != len(pf.Entries) {
		return nil, fmt.Errorf("numf: trailer declared %d detail records, parsed %d",
			declaredCount, len(pf.Entries))
	}
	return pf, nil
}

func NullStr(s string) any {
	if s == "" {
		return nil
	}
	return s
}

// FromFile reloads the data processing pipeline from one NUMF feed inside a
// transaction. Entries and data items are upserted since the same ones
// recur across every batch's feed; entries are always inserted fresh because an entry id is unique to its file and line.
func FromFile(ctx context.Context, pipeline *Pipeline, path string) (EntryCounts, error) {
	var c EntryCounts
	data, err := os.ReadFile(path)
	if err != nil {
		return c, err
	}
	pf, err := Process(data)
	if err != nil {
		return c, err
	}
	tx, err := pipeline.BeginTx(ctx, nil)
	if err != nil {
		return c, err
	}
	defer tx.Rollback()

	if pf.BatchCode != "" {
		if _, err := tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO batch_codes (batch_code, status, task_code)
			 VALUES (?, 'STARTED', ?)`, pf.BatchCode, pf.TaskCode); err != nil {
			return c, err
		}
		c.BatchCodes++
	}

	for i, ent := range pf.Entries {
		if ent.EntryID == "" || ent.DataID == "" {
			continue
		}
		res, err := tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO entries (entry_id, data_id, task_code) VALUES (?, ?, ?)`,
			ent.EntryID, pf.TaskCode)
		if err != nil {
			return c, err
		}
		if n, _ := res.RowsAffected(); n > 0 {
			c.EntryIDs++
		}
		res, err = tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO data_items (data_id, serial_number, entry_id)
			 VALUES (?, ?, ?)`, ent.DataID, fmt.Sprintf("SN%08d", i+1), ent.EntryID)
		if err != nil {
			return c, err
		}
		if n, _ := res.RowsAffected(); n > 0 {
			c.DataIDs++
		}
		entryID := fmt.Sprintf("ENT-%s-%04d", pf.BatchCode, i+1)
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO entries_data (entry_id, data_id, task_code, batch_code,
			        entry_type, entry_value, entry_timestamp, precision_flag)
			 VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
			entryID, ent.DataID, pf.TaskCode, nullStr(pf.BatchCode),
			ent.TypeCode, ent.Value, ent.Timestamp, ent.PrecisionFlag); err != nil {
			return c, err
		}
		c.Entries++
	}
	return c, tx.Commit()
}

// EnsureSeeded loads the seed only when the data processing pipeline is empty.
func EnsureSeeded(ctx context.Context, pipeline *Pipeline, path string) (bool, EntryCounts, error) {
	var n int
	if err := pipeline.QueryRowContext(ctx, `SELECT count(*) FROM entries_data`).Scan(&n); err != nil {
		return false, EntryCounts{}, err
	}
	if n != 0 {
		return false, EntryCounts{}, nil
	}
	c, err := FromFile(ctx, pipeline, path)
	return err == nil, c, err
}
