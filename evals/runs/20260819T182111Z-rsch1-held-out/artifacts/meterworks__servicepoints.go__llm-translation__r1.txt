package compute

type ServicePoint struct {
	ServicePointID string
	RouteCode      string
	AccountRef     string
	Meter          *Meter
}

type Meter struct {
	MeterID    string
	SerialNumber string
	SizeCode   string
}

type ReadSummary struct {
	ReadID           string
	MeterID          string
	ReadType         string
	ReadValue        float64
	ReadDate         string
	ToleranceFlag    bool
	ServicePointID string
}

func (s *ComputeService) GetServicePoint(ctx context.Context, spID string) (*ServicePoint, error) {
	var sp ServicePoint
	var mt Meter
	err := s.database.QueryRowContext(ctx, `SELECT service_point_id, route_code,
	              COALESCE(account_ref,''), meter_id, serial_number,
	              COALESCE(size_code,'')
	       FROM service_points WHERE service_point_id = ?`, spID).Scan(&sp.ServicePointID, &sp.RouteCode,
		&sp.AccountRef, &mt.MeterID, &mt.SerialNumber, &mt.SizeCode)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, fmt.Errorf("not_found: service point not found")
	}
	if err != nil {
		return nil, fmt.Errorf("server_error: %w", err)
	}
	sp.Meter = &mt
	return &sp, nil
}

// GetServicePointReads returns the read history at one service point,
// most recent first. Unlike the route listing, a service point that does
// not exist is a 404: it is a first-class row, not an inferred grouping.
func (s *ComputeService) GetServicePointReads(ctx context.Context, spID string, limit int, offset int) ([]ReadSummary, error) {
	var known string
	err := s.database.QueryRowContext(ctx,
		`SELECT service_point_id FROM service_points WHERE service_point_id = ?`, spID).Scan(&known)
	if errors.Is(err, sql.ErrNoRows) {
		return nil, fmt.Errorf("not_found: service point not found")
	}
	if err != nil {
		return nil, fmt.Errorf("server_error: %w", err)
	}

	rows, err := s.database.QueryContext(ctx, `SELECT read_id, meter_id, read_type,
	              read_value, read_date, tolerance_flag
	       FROM reads WHERE service_point_id = ?
	       ORDER BY read_date DESC LIMIT ? OFFSET ?`, spID, limit, offset)
	if err != nil {
		return nil, fmt.Errorf("server_error: %w", err)
	}
	defer rows.Close()
	out := []ReadSummary{}
	for rows.Next() {
		var it ReadSummary
		it.ServicePointID = spID
		if err := rows.Scan(&it.ReadID, &it.MeterID, &it.ReadType, &it.ReadValue,
			&it.ReadDate, &it.ToleranceFlag); err != nil {
			return nil, fmt.Errorf("server_error: %w", err)
		}
		out = append(out, it)
	}
	if err := rows.Err(); err != nil {
		return nil, fmt.Errorf("server_error: %w", err)
	}
	return out, nil
}
