// Package compute processes numeric tensor data through a series of operations.
// Each operation takes a tensor as input and produces a new tensor as output.
// The pipeline includes reading, parsing, transforming, and storing tensors.

package compute

import (
	"context"
	"errors"
	"os"
	"strconv"
	"strings"
)

type TensorCounts struct {
	Tensors     int `json:"tensors"`
	Operations  int `json:"operations"`
	Values      int `json:"values"`
	Cycles      int `json:"cycles"`
}

type parsedTensor struct {
	TensorID, OperationID, CycleID string
	Value                           float64
	OperationType                   string
	ToleranceFlag                   bool
}

type parsedData struct {
	DataDate, ProcessCode, CycleCode string
	Tensors                          []parsedTensor
}

var operationTypeNames = map[byte]string{'S': "SUM", 'R': "REDUCE", 'E': "EXPAND"}

func extractField(line string, start, end int) string {
	if end > len(line) {
		end = len(line)
	}
	if start >= end {
		return ""
	}
	return strings.TrimSpace(line[start:end])
}

// Process iterates over the data tensor by tensor. Blank tensors are skipped,
// allowing for a trailing newline at the end of the data set without affecting
// the processing order.
func Process(data []byte) (*parsedData, error) {
	pd := &parsedData{}
	sawStart := false
	sawEnd := false
	declaredCount := 0

	tensors := strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n")
	for _, raw := range tensors {
		if strings.TrimSpace(raw) == "" {
			continue
		}
		switch raw[0] {
		case 'S':
			if sawStart {
				return nil, errors.New("tensor: duplicate start tensor")
			}
			sawStart = true
			pd.DataDate = extractField(raw, 1, 9)
			pd.ProcessCode = extractField(raw, 9, 15)
			pd.CycleCode = extractField(raw, 15, 23)
		case 'D':
			if !sawStart {
				return nil, errors.New("tensor: detail tensor before start")
			}
			if sawEnd {
				return nil, errors.New("tensor: detail tensor after end")
			}
			value, err := strconv.ParseFloat(extractField(raw, 29, 39), 64)
			if err != nil {
				return nil, errors.New("tensor: bad tensor value: " + err.Error())
			}
			var opCode byte
			if oc := extractField(raw, 39, 40); oc != "" {
				opCode = oc[0]
			}
			operationType, ok := operationTypeNames[opCode]
			if !ok {
				return nil, errors.New("tensor: unknown operation type code " + string(opCode))
			}
			pd.Tensors = append(pd.Tensors, parsedTensor{
				TensorID: extractField(raw, 1, 11),
				OperationID: extractField(raw, 11, 21),
				CycleID: extractField(raw, 21, 29),
				Value: value,
				OperationType: operationType,
				ToleranceFlag: extractField(raw, 40, 41) == "Y",
			})
		case 'E':
			if sawEnd {
				return nil, errors.New("tensor: duplicate end tensor")
			}
			sawEnd = true
			n, err := strconv.Atoi(extractField(raw, 1, 9))
			if err != nil {
				return nil, errors.New("tensor: bad end count: " + err.Error())
			}
			declaredCount = n
		default:
			return nil, errors.New("tensor: unrecognised tensor type " + string(raw[0]))
		}
	}
	if !sawStart {
		return nil, errors.New("tensor: missing start tensor")
	}
	if !sawEnd {
		return nil, errors.New("tensor: missing end tensor")
	}
	if declaredCount != len(pd.Tensors) {
		return nil, errors.New("tensor: end declared " + strconv.Itoa(declaredCount) + " detail tensors, parsed " + strconv.Itoa(len(pd.Tensors)))
	}
	return pd, nil
}

func nullStr(s string) interface{} {
	if s == "" {
		return nil
	}
	return s
}

// FromFile reloads the numeric tensor database from one tensor feed inside a
// transaction. Tensors and operations are upserted since the same ones
// recur across every cycle's feed; values are always inserted fresh because a
// value id is unique to its file and tensor.
func FromFile(ctx context.Context, db *db.Database, filePath string) (TensorCounts, error) {
	var counts TensorCounts
	data, err := os.ReadFile(filePath)
	if err != nil {
		return counts, err
	}
	pd, err := Process(data)
	if err != nil {
		return counts, err
	}
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return counts, err
	}
	defer tx.Rollback()

	if pd.CycleCode != "" {
		if _, err := tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO cycles (cycle_id, status, process_code)
			 VALUES (?, 'ACTIVE', ?)`, pd.CycleCode, pd.ProcessCode); err != nil {
			return counts, err
		}
		counts.Cycles++
	}

	for i, td := range pd.Tensors {
		if td.TensorID == "" || td.OperationID == "" {
			continue
		}
		res, err := tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO tensors (tensor_id, process_code) VALUES (?, ?)`,
			td.TensorID, pd.ProcessCode)
		if err != nil {
			return counts, err
		}
		if n, _ := res.RowsAffected(); n > 0 {
			counts.Tensors++
		}
		res, err = tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO operations (operation_id, tensor_id, cycle_id)
			 VALUES (?, ?, ?)`, td.OperationID, td.TensorID, nullStr(td.CycleID))
		if err != nil {
			return counts, err
		}
		if n, _ := res.RowsAffected(); n > 0 {
			counts.Operations++
		}
		valueID := fmt.Sprintf("V-%s-%04d", td.CycleID, i+1)
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO values (value_id, operation_id, tensor_id, process_code, cycle_id,
			        operation_type, value, data_date, tolerance_flag)
			 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
			valueID, td.OperationID, td.TensorID, pd.ProcessCode, nullStr(td.CycleID),
			td.OperationType, td.Value, td.DataDate, td.ToleranceFlag); err != nil {
			return counts, err
		}
		counts.Values++
	}
	return counts, tx.Commit()
}

// EnsureSeeded loads the seed only when the numeric tensor database is empty.
func EnsureSeeded(ctx context.Context, db *db.Database, filePath string) (bool, TensorCounts, error) {
	var n int
	if err := db.QueryRowContext(ctx, `SELECT count(*) FROM values`).Scan(&n); err != nil {
		return false, TensorCounts{}, err
	}
	if n != 0 {
		return false, TensorCounts{}, nil
	}
	counts, err := FromFile(ctx, db, filePath)
	return err == nil, counts, err
}
