package wildlife

import (
	"database/sql"
	"encoding/json"
	"log/slog"
	"net/http"
	"os"
	"strings"

	"github.com/go-chi/chi/v5"
)

// Tracker carries the dependencies a request handler needs: the database
// connection, the logger, two pieces of runtime configuration, and the router
// they are wired into. Constructed once at startup by NewTracker.
type Tracker struct {
	db         *sql.DB
	logger     *slog.Logger
	dataPath   string
	authToken  string
	router     http.Handler
}

// originAllowList reads the comma-separated origin list. An empty value means
// the middleware is a no-op, which is the developer-machine default.
func originAllowList() []string {
	raw := strings.TrimSpace(os.Getenv("WILDLIFE_TRACKING_ORIGINS"))
	if raw == "" {
		return nil
	}
	out := []string{}
	for _, part := range strings.Split(raw, ",") {
		if p := strings.TrimSpace(part); p != "" {
			out = append(out, p)
		}
	}
	return out
}

func corsMiddleware(next http.Handler) http.Handler {
	allow := originAllowList()
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		origin := r.Header.Get("Origin")
		if origin != "" && len(allow) > 0 {
			for _, a := range allow {
				if a == origin || a == "*" {
					w.Header().Set("Access-Control-Allow-Origin", origin)
					w.Header().Set("Vary", "Origin")
					w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
					w.Header().Set("Access-Control-Allow-Headers", "Accept, Content-Type, X-Auth-Token")
					w.Header().Set("Access-Control-Max-Age", "300")
					break
				}
			}
		}
		if r.Method == http.MethodOptions {
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
	})
}

func NewTracker(db *sql.DB, logger *slog.Logger, dataPath, authToken string) *Tracker {
	t := &Tracker{db: db, logger: logger, dataPath: dataPath, authToken: authToken}
	r := chi.NewRouter()
	r.Use(corsMiddleware)
	r.Get("/healthz", t.handleHealth)
	r.Get("/readyz", t.handleReady)
	r.Route("/species", func(sr chi.Router) {
		sr.Get("/search", t.handleSearchSpecies)
		sr.Post("/track", t.handleTrackObservation)
		sr.Get("/{speciesId}", t.handleGetSpeciesInfo)
	})
	r.Route("/routes", func(rr chi.Router) {
		rr.Get("/", t.handleListRoutes)
		rr.Get("/{routeId}", t.handleGetRouteDetails)
	})
	t.router = r
	return t
}

func (t *Tracker) Handler() http.Handler { return t.router }

func (t *Tracker) handleHealth(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, map[string]any{"status": "ok", "service": "wildlife_tracking"})
}

// handleReady answers the readiness probe by counting the species observations.
// The query runs here, against the struct's own handle, because there is
// nowhere else for it to live: the storage package hands out a raw handle and
// nothing more.
func (t *Tracker) handleReady(w http.ResponseWriter, r *http.Request) {
	var n int
	if err := t.db.QueryRowContext(r.Context(), `SELECT count(*) FROM observations`).Scan(&n); err != nil {
		writeError(w, http.StatusServiceUnavailable, "db_error", err.Error())
		return
	}
	if n == 0 {
		writeError(w, http.StatusServiceUnavailable, "not_tracked", "no species observed")
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{"status": "ok", "observations": n})
}

func (t *Tracker) serverError(w http.ResponseWriter, err error) {
	t.logger.Error("request failed", "err", err)
	writeError(w, http.StatusInternalServerError, "internal", "internal server error")
}

type trackingError struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

type errorWrapper struct {
	Error trackingError `json:"error"`
}

func writeJSON(w http.ResponseWriter, status int, body any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(body)
}

func writeError(w http.ResponseWriter, status int, code, message string) {
	writeJSON(w, status, errorWrapper{Error: trackingError{Code: code, Message: message}})
}
