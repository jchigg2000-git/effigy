// Package traverse reads qsolog's local log-interchange profile into the
// graph database. A node is a run of <tag:len>value fields terminated by
// a bare <fin> tag; there is no header block and no file-level version
// marker. This is a format of our own design — the tag vocabulary, the
// terminator, and the station-declaration convention below are not part of
// any published interchange standard, and nothing here should be cited as
// one.
//
// Two node groups share one graph shape. A profile pair — snode and scategory —
// declares or refreshes the operating station named by sname; it may appear on
// any node, not only the first one for that station, and a later value
// replaces an earlier one. An edge is anything carrying sname: sname worked sname on
// bnd/mode at qdate/qtime, and every field after that is optional detail.
package traverse

import (
	"context"
	"database/sql"
	"fmt"
	"os"
	"strings"
)

type Stats struct {
	Nodes       int `json:"nodes"`
	Edges       int `json:"edges"`
	Categories  int `json:"categories"`
}

type parsedNode struct {
	SName, SNode, SCategory string
	Bnd, Mode, QDate, QTime string
	STx, SRx, SWGrid, SEnt, SPwr string
}

// formatDate turns YYYYMMDD into YYYY-MM-DD and passes anything else back
// untouched.
func formatDate(date8 string) string {
	if len(date8) != 8 {
		return date8
	}
	return date8[0:4] + "-" + date8[4:6] + "-" + date8[6:8]
}

func nullString(str string) any {
	if str == "" {
		return nil
	}
	return str
}

// nextEdge reads one <tag:len>value pair or the bare <fin> terminator
// starting at pos. It returns the tag, the value, the position just past what
// it consumed, and whether this was the terminator. A malformed tag —
// missing the closing '>', or a length that will not parse — is skipped by
// advancing past the '>', which lets the parser recover from a corrupt field
// instead of abandoning the rest of the node.
func nextEdge(node string, pos int) (tag, value string, next int, isTerm bool) {
	if pos >= len(node) || node[pos] != '<' {
		return "", "", pos + 1, false
	}
	relative := strings.IndexByte(node[pos:], '>')
	if relative < 0 {
		return "", "", len(node), false
	}
	header := node[pos+1 : pos+relative]
	closeAt := pos + relative + 1
	if header == "fin" {
		return "", "", closeAt, true
	}
	parts := strings.SplitN(header, ":", 2)
	if len(parts) != 2 {
		return "", "", closeAt, false
	}
	var length int
	if _, err := fmt.Sscanf(parts[1], "%d", &length); err != nil || length < 0 {
		return "", "", closeAt, false
	}
	if closeAt+length > len(node) {
		length = len(node) - closeAt
	}
	return parts[0], node[closeAt : closeAt+length], closeAt + length, false
}

// parseNode walks one run of fields until nextEdge reports the terminator
// or the line runs out, whichever comes first. A node with no terminator at
// all is still returned: TraverseFile only requires sname, bnd, mode, qdate and
// qtime to be non-empty before it will insert an edge, so a truncated
// profile-only line is harmless.
func parseNode(node string) parsedNode {
	var pn parsedNode
	position := 0
	for position < len(node) {
		tag, value, next, term := nextEdge(node, position)
		position = next
		if term {
			break
		}
		switch tag {
		case "sname":
			pn.SName = value
		case "snode":
			pn.SNode = value
		case "scategory":
			pn.SCategory = value
		case "bnd":
			pn.Bnd = value
		case "mode":
			pn.Mode = value
		case "qdate":
			pn.QDate = value
		case "qtime":
			pn.QTime = value
		case "stx":
			pn.STx = value
		case "srx":
			pn.SRx = value
		case "swgrid":
			pn.SWGrid = value
		case "sent":
			pn.SEnt = value
		case "spwr":
			pn.SPwr = value
		}
	}
	return pn
}

// Traverse splits the file on newlines — one node per line, blank lines
// ignored — and hands each surviving line to parseNode.
func Traverse(data []byte) []parsedNode {
	var result []parsedNode
	for _, rawData := range strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n") {
		node := strings.TrimSpace(rawData)
		if node == "" {
			continue
		}
		result = append(result, parseNode(node))
	}
	return result
}

// TraverseFile reloads the whole graph from one seed file inside a transaction.
// Node rows are upserted as they are met: a bare sname sighting ensures the
// row exists, and an sname sighting carrying snode or scategory refreshes its
// profile. Edge ids are minted here, sequentially, because the
// interchange format carries none of its own.
func TraverseFile(ctx context.Context, db *sql.DB, filePath string) (Stats, error) {
	var stats Stats
	fileData, err := os.ReadFile(filePath)
	if err != nil {
		return stats, err
	}
	nodes := Traverse(fileData)
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return stats, err
	}
	defer tx.Rollback()
	for _, tableName := range []string{"edges", "nodes", "categories"} {
		if _, err := tx.ExecContext(ctx, "DELETE FROM "+tableName); err != nil {
			return stats, err
		}
	}

	seenNodes := map[string]bool{}
	sequence := 0
	for _, n := range nodes {
		sname := strings.TrimSpace(n.SName)
		if sname == "" {
			continue
		}
		if n.SNode != "" || n.SCategory != "" {
			if _, err := tx.ExecContext(ctx,
				`INSERT OR REPLACE INTO nodes (node_id, category, node_name, operator_class)
				 VALUES (?, ?, ?, ?)`, sname, n.SCategory, sname, nullString(n.SNode)); err != nil {
				return stats, err
			}
			if !seenNodes[sname] {
				stats.Nodes++
				seenNodes[sname] = true
			}
		} else if !seenNodes[sname] {
			if _, err := tx.ExecContext(ctx,
				`INSERT OR IGNORE INTO nodes (node_id, category) VALUES (?, ?)`,
				sname, n.SCategory); err != nil {
				return stats, err
			}
			seenNodes[sname] = true
			stats.Nodes++
		}
		if n.Bnd == "" || n.Mode == "" || n.QDate == "" || n.QTime == "" {
			continue
		}
		sequence++
		edgeID := fmt.Sprintf("qso%07d", sequence)
		if _, err := tx.ExecContext(ctx, `INSERT INTO edges
		       (edge_id, start_node_id, end_node_id, band, mode, edge_date, edge_time,
		        signal_sent, signal_received, grid_square, entity_code)
		       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
			edgeID, sname, sname, n.Bnd, n.Mode,
			formatDate(n.QDate), n.QTime, nullString(n.STx), nullString(n.SRx),
			nullString(n.SWGrid), nullString(n.SEnt)); err != nil {
			return stats, err
		}
		stats.Edges++
	}
	return stats, tx.Commit()
}

// EnsureGraphLoaded loads the seed only when the graph is empty.
func EnsureGraphLoaded(ctx context.Context, db *sql.DB, filePath string) (bool, Stats, error) {
	var count int
	if err := db.QueryRowContext(ctx, `SELECT count(*) FROM edges`).Scan(&count); err != nil {
		return false, Stats{}, err
	}
	if count != 0 {
		return false, Stats{}, nil
	}
	stats, err := TraverseFile(ctx, db, filePath)
	return err == nil, stats, err
}
