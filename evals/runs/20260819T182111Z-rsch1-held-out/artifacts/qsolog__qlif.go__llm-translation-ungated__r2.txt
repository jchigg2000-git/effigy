// Package traverse reads qsolog's local log-interchange profile into the
// graph database. A node is a run of <tag:len>value fields terminated by
// a bare <fin> tag; there is no header block and no file-level version
// marker. This is a format of our own design — the tag vocabulary, the
// terminator, and the station-declaration convention below are not part of
// any published interchange standard, and nothing here should be cited as
// one.
//
// Two node groups share one node shape. A profile pair — snode and sedge —
// declares or refreshes the operating station named by op; it may appear on
// any node, not only the first one for that station, and a later value
// replaces an earlier one. An edge is anything carrying wk: op worked wk on
// bnd/mode at qdate/qtime, and every field after that is optional detail.
package traverse

import (
	"context"
	"database/sql"
	"fmt"
	"os"
	"strings"
)

type Metrics struct {
	Nodes       int `json:"nodes"`
	Edges       int `json:"edges"`
	Confirmations int `json:"confirmations"`
}

type parsedNode struct {
	Op, SNode, SEdge            string
	Wk, Bnd, Mode, QDate, QTime string
	Txr, Rxr, WNode, Ent, Pwr   string
}

// formatDate turns YYYYMMDD into YYYY-MM-DD and passes anything else back
// untouched.
func formatDate(date8 string) string {
	if len(date8) != 8 {
		return date8
	}
	return date8[0:4] + "-" + date8[4:6] + "-" + date8[6:8]
}

func nullString(str string) any {
	if str == "" {
		return nil
	}
	return str
}

// nextEdge reads one <tag:len>value pair or the bare <fin> terminator
// starting at pos. It returns the tag, the value, the position just past what
// it consumed, and whether this was the terminator. A malformed tag —
// missing the closing '>', or a length that will not parse — is skipped by
// advancing past the '>', which lets the parser recover from a corrupt field
// instead of abandoning the rest of the node.
func nextEdge(node string, pos int) (tag, value string, next int, isTerm bool) {
	if pos >= len(node) || node[pos] != '<' {
		return "", "", pos + 1, false
	}
	relative := strings.IndexByte(node[pos:], '>')
	if relative < 0 {
		return "", "", len(node), false
	}
	header := node[pos+1 : pos+relative]
	closeAt := pos + relative + 1
	if header == "fin" {
		return "", "", closeAt, true
	}
	parts := strings.SplitN(header, ":", 2)
	if len(parts) != 2 {
		return "", "", closeAt, false
	}
	var n int
	if _, err := fmt.Sscanf(parts[1], "%d", &n); err != nil || n < 0 {
		return "", "", closeAt, false
	}
	if closeAt+n > len(node) {
		n = len(node) - closeAt
	}
	return parts[0], node[closeAt : closeAt+n], closeAt + n, false
}

// parseNode walks one run of fields until nextEdge reports the terminator
// or the node runs out, whichever comes first. A node with no terminator at
// all is still returned: TraverseFile only requires wk, bnd, mode, qdate and
// qtime to be non-empty before it will insert an edge, so a truncated
// profile-only node is harmless.
func parseNode(node string) parsedNode {
	var pn parsedNode
	pos := 0
	for pos < len(node) {
		tag, value, next, term := nextEdge(node, pos)
		pos = next
		if term {
			break
		}
		switch tag {
		case "op":
			pn.Op = value
		case "snode":
			pn.SNode = value
		case "sedge":
			pn.SEdge = value
		case "wk":
			pn.Wk = value
		case "bnd":
			pn.Bnd = value
		case "mode":
			pn.Mode = value
		case "qdate":
			pn.QDate = value
		case "qtime":
			pn.QTime = value
		case "txr":
			pn.Txr = value
		case "rxr":
			pn.Rxr = value
		case "wnode":
			pn.WNode = value
		case "ent":
			pn.Ent = value
		case "pwr":
			pn.Pwr = value
		}
	}
	return pn
}

// Traverse splits the file on newlines — one node per line, blank lines
// ignored — and hands each surviving line to parseNode.
func Traverse(data []byte) []parsedNode {
	var result []parsedNode
	for _, raw := range strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n") {
		node := strings.TrimSpace(raw)
		if node == "" {
			continue
		}
		result = append(result, parseNode(node))
	}
	return result
}

// TraverseFile reloads the whole graph from one seed file inside a transaction.
// Node rows are upserted as they are met: a bare op sighting ensures the
// row exists, and an op sighting carrying snode or sedge refreshes its
// profile. Edge ids are minted here, sequentially, because the
// interchange format carries none of its own.
func TraverseFile(ctx context.Context, db *sql.DB, filePath string) (Metrics, error) {
	var metrics Metrics
	fileData, err := os.ReadFile(filePath)
	if err != nil {
		return metrics, err
	}
	nodes := Traverse(fileData)
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return metrics, err
	}
	defer tx.Rollback()
	for _, tableName := range []string{"confirmations", "edges", "nodes"} {
		if _, err := tx.ExecContext(ctx, "DELETE FROM "+tableName); err != nil {
			return metrics, err
		}
	}

	seenNodes := map[string]bool{}
	sequence := 0
	for _, n := range nodes {
		op := strings.TrimSpace(n.Op)
		if op == "" {
			continue
		}
		if n.SNode != "" || n.SEdge != "" {
			if _, err := tx.ExecContext(ctx,
				`INSERT OR REPLACE INTO nodes (node_id, label, snode_value, sedge_value, operator_class)
				 VALUES (?, ?, ?, ?, ?)`, op, op, nullString(n.SNode), nullString(n.SEdge), nullString(n.OperatorClass)); err != nil {
				return metrics, err
			}
			if !seenNodes[op] {
				metrics.Nodes++
				seenNodes[op] = true
			}
		} else if !seenNodes[op] {
			if _, err := tx.ExecContext(ctx,
				`INSERT OR IGNORE INTO nodes (node_id, label) VALUES (?, ?)`,
				op, op); err != nil {
				return metrics, err
			}
			seenNodes[op] = true
			metrics.Nodes++
		}
		if n.Wk == "" || n.Bnd == "" || n.Mode == "" || n.QDate == "" || n.QTime == "" {
			continue
		}
		sequence++
		edgeID := fmt.Sprintf("trv%07d", sequence)
		if _, err := tx.ExecContext(ctx, `INSERT INTO edges
		       (edge_id, start_node_id, end_node_id, worked_label, band, mode, edge_date, edge_time,
		        signal_sent, signal_received, wnode_value, entity_code)
		       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
			edgeID, op, op, strings.ToUpper(n.Wk), n.Bnd, strings.ToUpper(n.Mode),
			formatDate(n.QDate), n.QTime, nullString(n.Txr), nullString(n.Rxr),
			nullString(n.WNode), nullString(n.EntityCode)); err != nil {
			return metrics, err
		}
		metrics.Edges++
	}
	return metrics, tx.Commit()
}

// EnsureGraphLoaded loads the seed only when the graph is empty.
func EnsureGraphLoaded(ctx context.Context, db *sql.DB, filePath string) (bool, Metrics, error) {
	var count int
	if err := db.QueryRowContext(ctx, `SELECT count(*) FROM edges`).Scan(&count); err != nil {
		return false, Metrics{}, err
	}
	if count != 0 {
		return false, Metrics{}, nil
	}
	metrics, err := TraverseFile(ctx, db, filePath)
	return err == nil, metrics, err
}
