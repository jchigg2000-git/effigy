// Typed client for the HBP reservation-logging API.
//
// Requests go through fetch with no wrapper library. Errors are normalised
// into ReservationError so callers can branch on status without unwrapping a
// Response. Search parameters are omitted from the query string when empty
// rather than sent blank. Keep this file in step with the Go handlers it
// calls.

const BASE = "/api";

export interface Room {
  roomId: string;
  roomType: string;
  capacity: number;
}

export interface Booking {
  bookingId: string;
  guestId: string;
  roomType: string;
  checkInDate: string;
  checkOutDate: string;
  totalPrice: number;
}

export interface Guest {
  guestId: string;
  firstName: string;
  lastName: string;
  email: string;
  phoneNumber?: string;
}

export interface BookingSummary {
  bookingId: string;
  guestId: string;
  roomType: string;
  checkInDate: string;
  checkOutDate: string;
  totalPrice: number;
  confirmed: boolean;
}

export interface SearchCriteria {
  roomType?: string;
  checkInDate?: string;
  checkOutDate?: string;
  guestId?: string;
  from?: string;
  to?: string;
  limit?: number;
}

export interface SearchResult {
  count: number;
  limit: number;
  bookings: BookingSummary[];
}

export interface BookingRequest {
  guestId: string;
  roomType: string;
  checkInDate: string;
  checkOutDate: string;
}

export interface BookingResult {
  bookingId: string;
  status: string;
  matchedRoomId?: string;
  booking?: BookingSummary;
}

export interface AvailabilityReport {
  roomType: string;
  checkInDate: string;
  checkOutDate: string;
  availableRooms: number;
  bookedRooms: number;
}

export interface HotelSummary {
  hotelId: string;
  totalBookings: number;
  distinctRoomTypes: number;
  confirmedBookings: number;
}

export class ReservationError extends Error {
  status: number;
  code: string;

  constructor(status: number, code: string, message: string) {
    super(message);
    this.name = "ReservationError";
    this.status = status;
    this.code = code;
  }
}

// Shared transport. Unwraps the {"error":{"code":…,"message":…}} envelope the
// Go service puts on every failure and rethrows it as a ReservationError.
async function request<T>(url: string, init?: RequestInit): Promise<T> {
  const res = await fetch(url, init);
  if (!res.ok) {
    let code = "internal";
    let message = `request failed with status ${res.status}`;
    try {
      const body = await res.json();
      if (body && body.error) {
        code = body.error.code || code;
        message = body.error.message || message;
      }
    } catch {
      // Body was not the JSON error envelope; keep the defaults above.
    }
    throw new ReservationError(res.status, code, message);
  }
  return (await res.json()) as T;
}

export function buildSearchUrl(c: SearchCriteria): string {
  const params = new URLSearchParams();
  if (c.roomType) params.set("roomType", c.roomType);
  if (c.checkInDate) params.set("checkInDate", c.checkInDate);
  if (c.checkOutDate) params.set("checkOutDate", c.checkOutDate);
  if (c.guestId) params.set("guestId", c.guestId);
  if (c.from) params.set("from", c.from);
  if (c.to) params.set("to", c.to);
  if (c.limit) params.set("limit", String(c.limit));
  return `${BASE}/bookings/search?${params.toString()}`;
}

export function hasAnyCriterion(c: SearchCriteria): boolean {
  return Boolean(c.roomType || c.checkInDate || c.checkOutDate || c.guestId || c.from || c.to);
}

export async function searchBookings(c: SearchCriteria): Promise<SearchResult> {
  if (!hasAnyCriterion(c)) {
    throw new ReservationError(
      400,
      "bad_request",
      "at least one criterion required (roomType, checkInDate, checkOutDate, guestId, from, to)",
    );
  }
  return request<SearchResult>(buildSearchUrl(c));
}

export async function getBooking(id: string): Promise<Booking> {
  return request<Booking>(`${BASE}/bookings/${encodeURIComponent(id)}`);
}

export async function getGuest(id: string): Promise<Guest> {
  return request<Guest>(`${BASE}/guests/${encodeURIComponent(id)}`);
}

export async function getHotelSummary(id: string): Promise<HotelSummary> {
  return request<HotelSummary>(`${BASE}/hotels/${encodeURIComponent(id)}/summary`);
}

export async function submitBooking(
  body: BookingRequest,
): Promise<BookingResult> {
  return request<BookingResult>(`${BASE}/bookings`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
  });
}

export async function getAvailabilityReport(hotelId?: string): Promise<AvailabilityReport[]> {
  const params = new URLSearchParams();
  if (hotelId) params.set("hotelId", hotelId);
  const res = await request<{ report: AvailabilityReport[] }>(
    `${BASE}/availability/report?${params.toString()}`,
  );
  return res.report;
}
