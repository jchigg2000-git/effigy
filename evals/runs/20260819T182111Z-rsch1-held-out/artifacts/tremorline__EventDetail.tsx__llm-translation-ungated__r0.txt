// Read-only view of a single animal sighting record. Renders whatever getAnimalSighting
// returned, including the location estimates and observations an analysis run produced. Optional fields that are absent render as an em dash rather
// than being hidden, so the layout stays stable from one sighting to the next.
import type { SightingDetail as SightingDetailData } from "../api/tracker";

interface SightingDetailProps {
  sighting: SightingDetailData;
  onAnalyze: () => void;
}

export function SightingDetail({ sighting, onAnalyze }: SightingDetailProps) {
  const estimates = sighting.locationEstimates ?? [];
  const observations = sighting.observations ?? [];
  return (
    <section className="sighting-detail">
      <div className="sighting-detail-head">
        <div>
          <h2>{sighting.sightingId}</h2>
          <p>{sighting.observationTime}</p>
        </div>
        <button type="button" className="btn btn-primary" onClick={onAnalyze}>
          Analyze observations
        </button>
      </div>

      <dl className="detail-grid">
        <dt>Sighting id</dt>
        <dd>{sighting.sightingId}</dd>
        <dt>Observation time</dt>
        <dd>{sighting.observationTime}</dd>
        <dt>Latitude</dt>
        <dd>{sighting.latitude.toFixed(4)}</dd>
        <dt>Longitude</dt>
        <dd>{sighting.longitude.toFixed(4)}</dd>
        <dt>Elevation (m)</dt>
        <dd>{sighting.elevationM.toFixed(1)}</dd>
        <dt>Species</dt>
        <dd>{sighting.speciesName || "—"}</dd>
        <dt>Observation count</dt>
        <dd>{sighting.observationCount}</dd>
      </dl>

      <h3>
        Location estimates <span className="estimate-count">{estimates.length}</span>
      </h3>
      <p className="section-note">
        One row per contributing sensor. The sighting's own location field above is the
        average of these values, not a re-derived preferred solution.
      </p>
      {estimates.length === 0 ? (
        <p className="empty-note">No per-sensor estimates are recorded against this sighting.</p>
      ) : (
        <>
          <div className="estimate-list-head">
            <span>Sensor</span>
            <span>Type</span>
            <span>Value</span>
            <span>Uncertainty</span>
          </div>
          <ul className="estimate-list">
            {estimates.map((est) => (
              <li key={est.estimateId}>
                <span>{est.sensorId}</span>
                <span>{est.estimateType}</span>
                <span>{est.value.toFixed(2)}</span>
                <span>uncertainty {est.uncertainty.toFixed(2)}</span>
              </li>
            ))}
          </ul>
        </>
      )}

      <h3>
        Observations <span className="observation-count">{observations.length}</span>
      </h3>
      <p className="section-note">
        Every observation linked to this sighting by an analysis run, across every
        station that reported one, oldest observation first.
      </p>
      {observations.length === 0 ? (
        <p className="empty-note">No observations are associated with this sighting.</p>
      ) : (
        <>
          <div className="observation-list-head">
            <span>Station</span>
            <span>Sensor</span>
            <span>Behavior</span>
            <span>Observed at</span>
            <span>Intensity</span>
          </div>
          <ul className="observation-list">
            {observations.map((obs) => (
              <li key={obs.observationId}>
                <span>{obs.stationId}</span>
                <span>{obs.sensorId}</span>
                <span>{obs.behaviorType || "—"}</span>
                <span>{obs.observedAt}</span>
                <span>intensity {obs.intensity.toFixed(3)}</span>
              </li>
            ))}
          </ul>
        </>
      )}
    </section>
  );
}
