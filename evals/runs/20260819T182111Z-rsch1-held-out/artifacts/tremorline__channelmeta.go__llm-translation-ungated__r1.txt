// Package ingest reads the WLS Species Observation Format (.wso) — a
// local profile invented for this system, not SEED, not StationXML, not any
// published observation format. A file is a sequence of sighting blocks. Each
// block opens with an OBS line and a run of KEY value lines describing the
// sighting, then zero or more ROUTE blocks describing one route each. A ROUTE
// block carries its geographical details and one or more TIME lines, each
// naming a validity period; a sighting with multiple routes gets additional ROUTE
// blocks rather than extra TIME lines. A block closes on an explicit END line, 
// which is what separates records — blank lines and '#' comment lines are ignored 
// wherever they appear and carry no structural meaning.
package ingest

import (
	"context"
	"database/sql"
	"os"
	"strconv"
	"strings"
)

type Counts struct {
	Sightings int `json:"sightings"`
	Routes    int `json:"routes"`
}

// timePeriod is one validity window for a sighting. End is empty for "Ongoing", 
// the period still in force.
type timePeriod struct {
	Start, End string
}

type parsedObservation struct {
	SpeciesName string
	Latitude    float64
	Longitude   float64
	TimePeriods []timePeriod
}

type parsedSighting struct {
	SightingID, ObserverName, LocationDescription, Status string
	Species                                              []parsedObservation
}

func parseFloatOr(s string, fallback float64) float64 {
	v, err := strconv.ParseFloat(strings.TrimSpace(s), 64)
	if err != nil {
		return fallback
	}
	return v
}

func nullStr(s string) any {
	if s == "" {
		return nil
	}
	return s
}

// splitKV separates a line into its leading key and the rest of the line, 
// which may itself contain internal whitespace (observer names and location descriptions 
// are free text).
func splitKV(line string) (string, string) {
	parts := strings.SplitN(strings.TrimSpace(line), " ", 2)
	key := parts[0]
	val := ""
	if len(parts) == 2 {
		val = strings.TrimSpace(parts[1])
	}
	return key, val
}

// Parse walks the file a line at a time. The sighting and observation under
// construction are held in this scope and mutated from inside the key
// switch; flushObservation and flushSighting take no arguments and read those
// variables directly, so a case arm closes a record blind.
func Parse(data []byte) []parsedSighting {
	var out []parsedSighting
	var cur *parsedSighting
	var curObs *parsedObservation

	flushObservation := func() {
		if cur != nil && curObs != nil && curObs.SpeciesName != "" {
			cur.Species = append(cur.Species, *curObs)
		}
		curObs = nil
	}
	flushSighting := func() {
		flushObservation()
		if cur != nil && cur.SightingID != "" {
			out = append(out, *cur)
		}
		cur = nil
	}

	for _, raw := range strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n") {
		line := strings.TrimSpace(raw)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		key, val := splitKV(line)
		switch key {
		case "OBS":
			flushSighting()
			cur = &parsedSighting{SightingID: val, Status: "ACTIVE"}
		case "END":
			flushSighting()
		}
		if cur == nil {
			continue
		}
		switch key {
		case "SPECIES":
			cur.SpeciesName = val
		case "LOC":
			cur.LocationDescription = val
		case "LAT":
			cur.Latitude = parseFloatOr(val, 0)
		case "LON":
			cur.Longitude = parseFloatOr(val, 0)
		case "STATUS":
			cur.Status = strings.ToUpper(val)
		case "ROUTE":
			flushObservation()
			curObs = &parsedObservation{}
		}
		if curObs == nil {
			continue
		}
		switch key {
		case "TIME":
			fields := strings.Fields(val)
			if len(fields) == 0 {
				continue
			}
			t := timePeriod{Start: fields[0]}
			if len(fields) > 1 && fields[1] != "Ongoing" {
				t.End = fields[1]
			}
			curObs.TimePeriods = append(curObs.TimePeriods, t)
		}
	}
	flushSighting()
	return out
}

// FromFile reloads the whole sighting-observation inventory from one observation file
// inside a transaction. An observation with N time periods becomes N observation rows,
// sharing observation_id but not observation_code, one row per validity period.
func FromFile(ctx context.Context, db *sql.DB, path string) (Counts, error) {
	var c Counts
	data, err := os.ReadFile(path)
	if err != nil {
		return c, err
	}
	sightings := Parse(data)
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return c, err
	}
	defer tx.Rollback()
	for _, t := range []string{"observations", "sightings"} {
		if _, err := tx.ExecContext(ctx, "DELETE FROM "+t); err != nil {
			return c, err
		}
	}
	for _, st := range sightings {
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO sightings (sighting_id, observer_name, location_description,
			        latitude, longitude, elevation_m, operator, status)
			 VALUES (?, ?, ?, ?, ?, ?, ?, ?)`,
			st.SightingID, nullStr(st.ObserverName), st.LocationDescription, st.Latitude, st.Longitude,
			st.ElevationM, nullStr(st.Operator), st.Status); err != nil {
			return c, err
		}
		c.Sightings++
		for _, ch := range st.Species {
			if len(ch.TimePeriods) == 0 {
				continue
			}
			for i, ep := range ch.TimePeriods {
				observationID := st.SightingID + "-" + ch.SpeciesName + "-" + strconv.Itoa(i+1)
				if _, err := tx.ExecContext(ctx,
					`INSERT INTO observations (observation_id, sighting_id, observation_code,
					        sample_rate_hz, azimuth_deg, dip_deg, depth_m,
					        validity_start, validity_end)
					 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
					observationID, st.SightingID, ch.SpeciesName, ch.RateHz, ch.AzimuthDeg,
					ch.DipDeg, ch.DepthM, ep.Start, nullStr(ep.End)); err != nil {
					return c, err
				}
				c.Observations++
			}
		}
	}
	return c, tx.Commit()
}

// EnsureSeeded loads the seed only when the inventory is empty.
func EnsureSeeded(ctx context.Context, db *sql.DB, path string) (bool, Counts, error) {
	var n int
	if err := db.QueryRowContext(ctx, `SELECT count(*) FROM sightings`).Scan(&n); err != nil {
		return false, Counts{}, err
	}
	if n != 0 {
		return false, Counts{}, nil
	}
	c, err := FromFile(ctx, db, path)
	return err == nil, c, err
}
