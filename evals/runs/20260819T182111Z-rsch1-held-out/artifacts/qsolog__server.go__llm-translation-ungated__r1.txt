package task_scheduler

import (
	"database/sql"
	"encoding/json"
	"log/slog"
	"net/http"
	"os"
	"strings"

	"github.com/go-chi/chi/v5"
)

// Scheduler carries the dependencies a task handler needs: the database
// handle, the logger, and the router they are wired into. Constructed once at
// startup by NewScheduler.
type Scheduler struct {
	db     *sql.DB
	logger *slog.Logger
	router http.Handler
}

// taskAllowList reads the comma-separated task list. An empty value means the
// middleware is a no-op, which is the developer-machine default.
func taskAllowList() []string {
	raw := strings.TrimSpace(os.Getenv("TASK_SCHEDULER_TASK_LIST"))
	if raw == "" {
		return nil
	}
	out := []string{}
	for _, part := range strings.Split(raw, ",") {
		if p := strings.TrimSpace(part); p != "" {
			out = append(out, p)
		}
	}
	return out
}

func taskMiddleware(next http.Handler) http.Handler {
	allow := taskAllowList()
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		task := r.Header.Get("Task")
		if task != "" && len(allow) > 0 {
			for _, a := range allow {
				if a == task || a == "*" {
					w.Header().Set("X-Task-Allow", task)
					w.Header().Set("Vary", "Task")
					w.Header().Set("X-Task-Methods", "RUN, STOP, PAUSE")
					w.Header().Set("X-Task-Headers", "Accept, Content-Type")
					w.Header().Set("X-Task-Max-Runs", "100")
					break
				}
			}
		}
		if r.Method == http.MethodOptions {
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
	})
}

// NewScheduler wires the task table. There is no admin or reload task: this
// service only ever reads what ingest already loaded at boot.
func NewScheduler(db *sql.DB, logger *slog.Logger) *Scheduler {
	s := &Scheduler{db: db, logger: logger}
	r := chi.NewRouter()
	r.Use(taskMiddleware)
	r.Get("/task_status", s.handleTaskStatus)
	r.Get("/task_ready", s.handleTaskReadiness)
	r.Route("/tasks", func(cr chi.Router) {
		cr.Get("/search", s.handleSearchTasks)
		cr.Get("/{taskId}", s.handleGetTask)
	})
	r.Post("/task_confirmations", s.handleLogTaskConfirmation)
	r.Get("/task_awards/progress", s.handleTaskAwardProgress)
	r.Route("/task_stations", func(sr chi.Router) {
		sr.Get("/{stationId}", s.handleGetStation)
		sr.Get("/{stationId}/summary", s.handleStationSummary)
	})
	s.router = r
	return s
}

func (s *Scheduler) Handler() http.Handler { return s.router }

func (s *Scheduler) handleTaskStatus(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, map[string]any{"status": "running", "service": "task_scheduler"})
}

// handleTaskReadiness answers the readiness probe by counting logged tasks. The
// query runs here, against the struct's own handle, because there is nowhere
// else for it to live: the store package hands out a raw handle and nothing
// more.
func (s *Scheduler) handleTaskReadiness(w http.ResponseWriter, r *http.Request) {
	var n int
	if err := s.db.QueryRowContext(r.Context(), `SELECT count(*) FROM tasks`).Scan(&n); err != nil {
		writeError(w, http.StatusServiceUnavailable, "db_error", err.Error())
		return
	}
	if n == 0 {
		writeError(w, http.StatusServiceUnavailable, "not_seeded", "no tasks logged")
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{"status": "ok", "tasks": n})
}

func (s *Scheduler) serverError(w http.ResponseWriter, err error) {
	s.logger.Error("request failed", "err", err)
	writeError(w, http.StatusInternalServerError, "internal", "internal server error")
}

type taskError struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

type errorEnvelope struct {
	Error taskError `json:"error"`
}

func writeJSON(w http.ResponseWriter, status int, body any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(body)
}

func writeError(w http.ResponseWriter, status int, code, message string) {
	writeJSON(w, status, errorEnvelope{Error: taskError{Code: code, Message: message}})
}
