// Read-only view of a single user profile. Renders the record returned by
// getUserProfile, including every track that has been listened to by the user.
// Optional fields that are absent render as an em dash rather than being
// hidden, so the layout stays stable from one record to the next.
import type { UserProfile } from "../api/user";

interface UserProfileDetailProps {
  userProfile: UserProfile;
}

export function UserProfileDetail({ userProfile }: UserProfileDetailProps) {
  const tracksListened = userProfile.tracksListened ?? [];
  const active = Boolean(userProfile.lastActive);
  return (
    <section className="user-profile-detail">
      <div className="user-profile-detail-head">
        <div>
          <h2>{userProfile.displayName}</h2>
          <p>{userProfile.userId}</p>
        </div>
        <span className={active ? "status-active" : "status-inactive"}>
          {active ? "Active" : "Inactive"}
        </span>
      </div>

      <dl className="detail-grid">
        <dt>User id</dt>
        <dd>{userProfile.userId}</dd>
        <dt>Country</dt>
        <dd>{userProfile.country ? userProfile.country.name : userProfile.countryCode}</dd>
        <dt>Display name</dt>
        <dd>{userProfile.displayName}</dd>
        <dt>Subscription plan</dt>
        <dd>{userProfile.subscriptionPlan}</dd>
        <dt>Account created</dt>
        <dd>{userProfile.accountCreated}</dd>
        <dt>Last login</dt>
        <dd>{userProfile.lastLogin || "—"}</dd>
        <dt>Favorite genre</dt>
        <dd>{userProfile.favoriteGenre || "—"}</dd>
        <dt>Listening hours</dt>
        <dd>{userProfile.listeningHours || "—"}</dd>
        <dt>Track count</dt>
        <dd>{userProfile.trackCount || "—"}</dd>
      </dl>

      <h3>
        Tracks Listened To <span className="track-listened-count">{tracksListened.length}</span>
      </h3>
      {tracksListened.length === 0 ? (
        <p className="empty-note">
          No tracks have been listened to by this user yet. A track only
          appears here once it has been played back by the user.
        </p>
      ) : (
        <ul className="track-listened-list">
          {tracksListened.map((t) => (
            <li key={t.trackId}>
              <span>{t.artistName}</span>
              <span>{t.trackTitle}</span>
              <span>{t.duration}</span>
              <span>{t.listenedAt}</span>
              <span>{t.listenStatus}</span>
            </li>
          ))}
        </ul>
      )}

      {/* The playlist progress rollup lives at the user level, not here: one
          user's listening activity says nothing about how many distinct playlists
          a user has added to their library. */}
      <p className="hint">
        Playlist progress credit for this user, if any, is reflected in the user's
        playlist progress rollup rather than repeated on this page.
      </p>
    </section>
  );
}
