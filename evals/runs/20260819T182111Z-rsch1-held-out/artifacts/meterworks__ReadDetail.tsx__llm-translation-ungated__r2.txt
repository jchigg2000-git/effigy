// Read-only view of a single playback session. Renders the record returned by getSessionDetails,
// including the listening duration computed against the user's prior
// trusted session when one exists. Optional fields render as an em dash rather than being hidden, so the layout stays stable from one session to the next.
import type { PlaybackSession } from "../api/player";

interface SessionDetailProps {
  session: PlaybackSession;
  onReport: () => void;
}

function formatMinutes(minutes: number): string {
  return `${minutes.toFixed(2)} min`;
}

// The server returns a total listen time, not a per-track rate; the average shown
// here is derived client-side purely for display and is never sent back.
function formatAverageTrackDuration(listenTimeMinutes: number, tracksListened: number): string {
  if (tracksListened <= 0) {
    return "—";
  }
  return `${formatMinutes(Math.round(listenTimeMinutes / tracksListened))} / track`;
}

const SESSION_STATUS_NOTES: Record<string, string> = {
  ACTIVE: "Playing on the main feed.",
  PAUSED: "Paused during a break; not a continuous play.",
  SKIPPED: "No audio collected; excluded from listening durations.",
};

export function SessionDetail({ session, onReport }: SessionDetailProps) {
  const user = session.user;
  const listening = session.listening;
  const statusNote = SESSION_STATUS_NOTES[session.sessionStatus] ?? "";
  return (
    <section className="session-detail">
      <div className="session-detail-head">
        <div>
          <h2>{session.sessionId}</h2>
          <p>{user?.userId}</p>
        </div>
        <button type="button" className="btn btn-primary" onClick={onReport}>
          Report / reschedule
        </button>
      </div>

      <dl className="detail-grid">
        <dt>Session id</dt>
        <dd>{session.sessionId}</dd>
        <dt>User id</dt>
        <dd>{session.userId}</dd>
        <dt>Playlist</dt>
        <dd>{session.playlistId}</dd>
        <dt>Listen cycle</dt>
        <dd>{session.listenCycleId}</dd>
        <dt>Session status</dt>
        <dd>{session.sessionStatus}</dd>
        <dt>Track played</dt>
        <dd>{session.trackPlayedId}</dd>
        <dt>Listen start</dt>
        <dd>{session.listenStartDate}</dd>
        <dt>Listen tolerance</dt>
        <dd>{session.toleranceFlag ? "flagged" : "within tolerance"}</dd>
        <dt>Playback issue</dt>
        <dd>{session.playbackIssue || "—"}</dd>
        <dt>User account</dt>
        <dd>{user?.accountRef || "—"}</dd>
        <dt>Device model</dt>
        <dd>{user?.device?.model || "—"}</dd>
        <dt>Device type</dt>
        <dd>{user?.device?.typeCode || "—"}</dd>
      </dl>

      {statusNote !== "" && <p className="hint">{statusNote}</p>}

      <h3>Listening duration</h3>
      {listening === undefined ? (
        <p className="empty-note">
          No prior trusted session on this device to measure against. The duration appears
          once a second active or paused session exists after this one.
        </p>
      ) : (
        <div className="shelf-grid">
          <div>
            <span>Prior session</span>
            <span>{listening.priorSessionId || "—"}</span>
          </div>
          <div>
            <span>Tracks listened</span>
            <span>{listening.tracksListened}</span>
          </div>
          <div>
            <span>Charged minutes</span>
            <span>{formatMinutes(listening.chargedMinutes)}</span>
          </div>
          <div>
            <span>Average track duration</span>
            <span>{formatAverageTrackDuration(listening.chargedMinutes, listening.tracksListened)}</span>
          </div>
        </div>
      )}
    </section>
  );
}
