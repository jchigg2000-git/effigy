// Main screen: animal search on the left, and either the selected animal's
// record or the confirmation form on the right. All request orchestration and
// all user-facing error text live in this file; this domain has no separate
// sightings-list or search-form component, unlike the library-catalog corpus.
import { useState } from "react";
import {
  ApiError,
  getAnimal,
  searchAnimals,
  submitObservation,
  type ObservationRequest,
  type ObservationResult,
  type Animal,
  type AnimalSummary,
  type SearchCriteria,
} from "./api/tracking";
import { ObservationForm } from "./components/ObservationForm";
import { AnimalDetail } from "./components/AnimalDetail";

const DEFAULT_CRITERIA: SearchCriteria = {
  habitat: "",
  behavior: "",
  locationId: "",
  startDate: "",
  endDate: "",
  limit: 50,
};

export function WildlifeApp() {
  const [criteria, setCriteria] = useState<SearchCriteria>(DEFAULT_CRITERIA);
  const [sightings, setSightings] = useState<AnimalSummary[]>([]);
  const [total, setTotal] = useState<number | null>(null);
  const [selected, setSelected] = useState<Animal | null>(null);
  const [observing, setObserving] = useState(false);
  const [observationResult, setObservationResult] = useState<ObservationResult | null>(null);
  const [fetching, setFetching] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  const explainError = (err: unknown): string => {
    if (err instanceof ApiError) {
      return `${err.code}: ${err.message}`;
    }
    return "the request could not be completed";
  };

  const performSearch = async () => {
    setFetching(true);
    setErrorMessage(null);
    setSelected(null);
    setObserving(false);
    try {
      const response = await searchAnimals(criteria);
      setSightings(response.animals);
      setTotal(response.total);
    } catch (err) {
      setSightings([]);
      setTotal(null);
      setErrorMessage(explainError(err));
    } finally {
      setFetching(false);
    }
  };

  const viewAnimal = async (animalId: string) => {
    setFetching(true);
    setErrorMessage(null);
    setObserving(false);
    setObservationResult(null);
    try {
      setSelected(await getAnimal(animalId));
    } catch (err) {
      setSelected(null);
      setErrorMessage(explainError(err));
    } finally {
      setFetching(false);
    }
  };

  const processObservation = async (req: ObservationRequest) => {
    setErrorMessage(null);
    try {
      const outcome = await submitObservation(req);
      setObservationResult(outcome);
      setObserving(false);
      if (outcome.matchedAnimalId) {
        setSelected(await getAnimal(outcome.matchedAnimalId));
      }
    } catch (err) {
      setErrorMessage(explainError(err));
    }
  };

  const updateText = (key: keyof SearchCriteria, value: string) => {
    setCriteria({ ...criteria, [key]: value });
  };

  return (
    <div className="wildlife-app-shell">
      <header className="wildlife-app-header">
        <div>
          <h1>qsolog</h1>
          <p>Animal sightings, observations, and conservation progress</p>
        </div>
        <button type="button" className="btn btn-primary" onClick={() => setObserving(true)}>
          Record an observation
        </button>
      </header>

      {errorMessage !== null && <p className="error-banner">{errorMessage}</p>}
      {observationResult !== null && (
        <p className="result-count">observation {observationResult.observationId}: {observationResult.matchStatus}</p>
      )}

      <main className="wildlife-app-body">
        <section className="wildlife-column-left">
          <form
            className="search-form"
            onSubmit={(e) => {
              e.preventDefault();
              performSearch();
            }}
          >
            <h2>Locate animals</h2>
            <label htmlFor="c-habitat">Habitat</label>
            <input id="c-habitat" value={criteria.habitat ?? ""} onChange={(e) => updateText("habitat", e.target.value)} />
            <label htmlFor="c-behavior">Behavior</label>
            <input id="c-behavior" value={criteria.behavior ?? ""} onChange={(e) => updateText("behavior", e.target.value)} />
            <label htmlFor="c-start">Start date</label>
            <input id="c-start" value={criteria.startDate ?? ""} onChange={(e) => updateText("startDate", e.target.value)} />
            <label htmlFor="c-end">End date</label>
            <input id="c-end" value={criteria.endDate ?? ""} onChange={(e) => updateText("endDate", e.target.value)} />
            <button type="submit" className="btn btn-primary" disabled={fetching}>
              {fetching ? "Searching…" : "Search"}
            </button>
          </form>

          {total !== null && <p className="result-count">{total} animals located</p>}
          {sightings.length === 0 ? (
            <p className="empty-note">No animals match those criteria.</p>
          ) : (
            <table className="sightings-table">
              <thead>
                <tr>
                  <th>Sighting id</th>
                  <th>Habitat</th>
                  <th>Behavior</th>
                  <th>Date</th>
                  <th>Observed</th>
                </tr>
              </thead>
              <tbody>
                {sightings.map((sighting) => (
                  <tr
                    key={sighting.sightingId}
                    className={sighting.sightingId === selected?.sightingId ? "row-selected" : ""}
                    onClick={() => viewAnimal(sighting.sightingId)}
                  >
                    <td>{sighting.sightingId}</td>
                    <td>{sighting.habitat}</td>
                    <td>{sighting.behavior}</td>
                    <td>{sighting.sightingDate}</td>
                    <td>{sighting.observed ? "yes" : "no"}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}
        </section>

        <section className="wildlife-column-right">
          {observing && <ObservationForm onSubmit={processObservation} onCancel={() => setObserving(false)} />}
          {!observing && selected === null && <p className="empty-note">Choose an animal to view its details.</p>}
          {!observing && selected !== null && <AnimalDetail animal={selected} />}
        </section>
      </main>
    </div>
  );
}
