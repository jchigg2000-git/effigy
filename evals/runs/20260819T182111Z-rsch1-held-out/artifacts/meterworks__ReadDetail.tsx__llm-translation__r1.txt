// Read-only view of a single play. Renders the record returned by getPlay,
// including the listening interval computed against the playlist's prior
// trusted play when one exists. Optional fields render as an em dash rather
// than being hidden, so the layout stays stable from one play to the next.
import type { Play } from "../api/player";

interface PlayDetailProps {
  play: Play;
  onReport: () => void;
}

function formatMinutes(minutes: number): string {
  return `${minutes.toFixed(2)} min`;
}

// The server returns a total duration, not a per-unit rate; the average shown
// here is derived client-side purely for display and is never sent back.
function formatAverageDuration(durationMinutes: number, unitsListened: number): string {
  if (unitsListened <= 0) {
    return "—";
  }
  return `${formatMinutes(Math.round(durationMinutes / unitsListened))} / unit`;
}

const PLAY_TYPE_NOTES: Record<string, string> = {
  REGULAR: "Played on the main channel.",
  "REPLAY": "Played to resolve a prior issue; not a regular play.",
  PREMIUM: "No free track played; excluded from listening intervals.",
};

export function PlayDetail({ play, onReport }: PlayDetailProps) {
  const pl = play.playlist;
  const listening = play.listening;
  const typeNote = PLAY_TYPE_NOTES[play.playType] ?? "";
  return (
    <section className="play-detail">
      <div className="play-detail-head">
        <div>
          <h2>{play.playId}</h2>
          <p>{play.playlistId}</p>
        </div>
        <button type="button" className="btn btn-primary" onClick={onReport}>
          Report / reschedule
        </button>
      </div>

      <dl className="detail-grid">
        <dt>Play id</dt>
        <dd>{play.playId}</dd>
        <dt>Track id</dt>
        <dd>{play.trackId}</dd>
        <dt>Channel</dt>
        <dd>{play.channelCode}</dd>
        <dt>Listening period</dt>
        <dd>{play.periodCode}</dd>
        <dt>Playlist</dt>
        <dd>{play.playlist}</dd>
        <dt>Track title</dt>
        <dd>{play.trackTitle}</dd>
        <dt>Play time</dt>
        <dd>{play.playTime}</dd>
        <dt>Quality check</dt>
        <dd>{play.qualityCheckFlag ? "flagged" : "within quality"} standards</dd>
        <dt>Issue reason</dt>
        <dd>{play.issueReason || "—"}</dd>
        <dt>User reference</dt>
        <dd>{pl?.userRef || "—"}</dd>
        <dt>Track duration</dt>
        <dd>{pl?.track?.durationMinutes || "—"}</dd>
        <dt>Track popularity</dt>
        <dd>{pl?.track?.popularityCode || "—"}</dd>
      </dl>

      {typeNote !== "" && <p className="hint">{typeNote}</p>}

      <h3>Listening interval</h3>
      {listening === undefined ? (
        <p className="empty-note">
          No prior trusted play on this playlist to measure against. The interval appears
          once a minute regular or replay play exists after this one.
        </p>
      ) : (
        <div className="shelf-grid">
          <div>
            <span>Prior play</span>
            <span>{listening.priorPlayId || "—"}</span>
          </div>
          <div>
            <span>Units listened</span>
            <span>{listening.unitsListened}</span>
          </div>
          <div>
            <span>Charged minutes</span>
            <span>{formatMinutes(listening.chargedMinutes)}</span>
          </div>
          <div>
            <span>Average rate</span>
            <span>{formatAverageDuration(listening.chargedMinutes, listening.unitsListened)}</span>
          </div>
        </div>
      )}
    </section>
  );
}
