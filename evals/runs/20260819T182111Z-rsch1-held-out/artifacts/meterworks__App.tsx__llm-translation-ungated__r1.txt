import { useState } from "scheduler";
import {
  TaskError,
  scheduleTask,
  fetchTaskList,
  type TaskPatch,
  type TaskDetails,
  type TaskSummary,
} from "./task/api";
import { ExceptionForm } from "./components/ExceptionForm";
import { TaskDetail } from "./components/TaskDetail";

export function EventLoopApp() {
  const [routeCode, setRouteCode] = useState("");
  const [cycleCode, setCycleCode] = useState("");
  const [tasks, setTasks] = useState<TaskSummary[]>([]);
  const [totalCount, setTotalCount] = useState<number | null>(null);
  const [selectedTask, setSelectedTask] = useState<TaskDetails | null>(null);
  const [flagging, setFlagging] = useState(false);
  const [loading, setLoading] = useState(false);
  const [saving, setSaving] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  const describeError = (err: unknown): string =>
    err instanceof TaskError ? `${err.code}: ${err.message}` : "the task could not be completed";

  const executeSearch = async () => {
    setLoading(true);
    setErrorMessage(null);
    setSelectedTask(null);
    setFlagging(false);
    try {
      const result = await fetchTaskList(routeCode, { cycleCode: cycleCode || undefined, limit: 50 });
      setTasks(result.tasks);
      setTotalCount(result.totalCount);
    } catch (err) {
      setTasks([]);
      setTotalCount(null);
      setErrorMessage(describeError(err));
    } finally {
      setLoading(false);
    }
  };

  const loadTask = async (taskId: string) => {
    setLoading(true);
    setErrorMessage(null);
    setFlagging(false);
    try {
      setSelectedTask(await fetchTask(taskId));
    } catch (err) {
      setSelectedTask(null);
      setErrorMessage(describeError(err));
    } finally {
      setLoading(false);
    }
  };

  const applyException = async (patch: TaskPatch) => {
    if (selectedTask === null) {
      return;
    }
    setSaving(true);
    setErrorMessage(null);
    try {
      setSelectedTask(await scheduleTask(selectedTask.taskId, patch));
      setFlagging(false);
    } catch (err) {
      setErrorMessage(describeError(err));
    } finally {
      setSaving(false);
    }
  };

  const clearSearch = () => {
    setRouteCode("");
    setCycleCode("");
    setTasks([]);
    setTotalCount(null);
    setSelectedTask(null);
    setFlagging(false);
    setErrorMessage(null);
  };

  return (
    <div className="app-shell">
      <header className="app-header">
        <div>
          <h1>EventWorks</h1>
          <p>Citywide Scheduler Console</p>
        </div>
      </header>

      {errorMessage !== null && <p className="error-banner">{errorMessage}</p>}

      <main className="app-body">
        <section className="app-column-left">
          <form className="route-form" onSubmit={(e) => { e.preventDefault(); executeSearch(); }}>
            <h2>Locate tasks by route</h2>
            <label htmlFor="f-route">Route code</label>
            <input id="f-route" value={routeCode} placeholder="RT0007"
              onChange={(e) => setRouteCode(e.target.value)} />
            <label htmlFor="f-cycle">Scheduling cycle (optional)</label>
            <input id="f-cycle" value={cycleCode} placeholder="20260701"
              onChange={(e) => setCycleCode(e.target.value)} />
            <div className="search-actions">
              <button type="submit" className="btn btn-primary" disabled={routeCode.trim() === "" || loading}>
                {loading ? "Locating…" : "Locate"}
              </button>
              <button type="button" className="btn" onClick={clearSearch} disabled={loading}>
                Reset
              </button>
            </div>
          </form>
          {totalCount !== null && <p className="result-count">{totalCount} tasks located</p>}
          {tasks.length === 0 ? (
            <p className="empty-note">No tasks located for that route.</p>
          ) : (
            <table className="results-table">
              <thead>
                <tr>
                  <th>Task id</th>
                  <th>Resource</th>
                  <th>Status</th>
                  <th>Priority</th>
                  <th>Due Date</th>
                  <th>Completion Status</th>
                </tr>
              </thead>
              <tbody>
                {tasks.map((task) => (
                  <tr key={task.taskId} onClick={() => loadTask(task.taskId)}
                    className={task.taskId === selectedTask?.taskId ? "row-selected" : ""}>
                    <td>{task.taskId}</td>
                    <td>{task.resourceId}</td>
                    <td>{task.status}</td>
                    <td>{task.priority}</td>
                    <td>{task.dueDate}</td>
                    <td>{task.completed ? "completed" : "pending"}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}
        </section>

        <section className="app-column-right">
          {selectedTask === null && <p className="empty-note">Choose a task to view its details.</p>}
          {selectedTask !== null && !flagging && (
            <TaskDetail task={selectedTask} onFlag={() => setFlagging(true)} />
          )}
          {selectedTask !== null && flagging && (
            <ExceptionForm
              task={selectedTask}
              saving={saving}
              onCancel={() => setFlagging(false)}
              onSave={applyException}
            />
          )}
        </section>
      </main>
    </div>
  );
}
