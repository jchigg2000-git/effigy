package api

import (
	"net/http"
	"strings"

	"github.com/go-chi/chi/v5"

	"tremorline/internal/model"
)

type playlistDetails struct {
	PlaylistID string `json:"playlistId"`
	UserID     string `json:"userId"`
	Title      string `json:"title"`
	Description string `json:"description"`
	TrackCount int    `json:"trackCount"`
}

// handleListPlaylists responds with a list of playlists, including the number of tracks in each. Filters by userID and title are optional.
func (s *Server) handleListPlaylists(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	conds := []string{}
	args := []any{}
	if v := strings.TrimSpace(q.Get("userID")); v != "" {
		conds = append(conds, "pl.user_id = ?")
		args = append(args, strings.ToUpper(v))
	}
	if v := strings.TrimSpace(q.Get("title")); v != "" {
		conds = append(conds, "pl.title LIKE ?")
		args = append(args, "%"+v+"%")
	}
	where := ""
	if len(conds) > 0 {
		where = "WHERE " + strings.Join(conds, " AND ")
	}
	rows, err := s.db.QueryContext(r.Context(), `SELECT pl.playlist_id, pl.user_id,
	                  pl.title, pl.description, count(t.track_id)
	           FROM playlists pl LEFT JOIN tracks t ON t.playlist_id = pl.playlist_id
	           `+where+`
	           GROUP BY pl.playlist_id ORDER BY pl.playlist_id`, args...)
	if err != nil {
		s.serverError(w, err)
		return
	}
	defer rows.Close()
	out := []playlistDetails{}
	for rows.Next() {
		var pl playlistDetails
		if err := rows.Scan(&pl.PlaylistID, &pl.UserID, &pl.Title, &pl.Description,
			&pl.TrackCount); err != nil {
			s.serverError(w, err)
			return
		}
		out = append(out, pl)
	}
	if err := rows.Err(); err != nil {
		s.serverError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"count":    len(out),
		"playlists": out,
	})
}

// handleGetPlaylistTracks retrieves a specific playlist along with all the tracks it contains, ordered by track ID.
func (s *Server) handleGetPlaylistTracks(w http.ResponseWriter, r *http.Request) {
	playlistID := strings.TrimSpace(chi.URLParam(r, "playlistId"))
	var pl model.Playlist
	err := s.db.QueryRowContext(r.Context(), `SELECT playlist_id, user_id, title,
	              description
	       FROM playlists WHERE playlist_id = ?`, playlistID).
		Scan(&pl.PlaylistID, &pl.UserID, &pl.Title, &pl.Description)
	if err != nil {
		writeError(w, http.StatusNotFound, "not_found", "playlist not found")
		return
	}
	rows, err := s.db.QueryContext(r.Context(), `SELECT track_id, playlist_id, title,
	                  artist, duration_ms, release_date
	           FROM tracks WHERE playlist_id = ?
	           ORDER BY track_id`, playlistID)
	if err != nil {
		s.serverError(w, err)
		return
	}
	defer rows.Close()
	tracks := []model.Track{}
	for rows.Next() {
		var t model.Track
		if err := rows.Scan(&t.TrackID, &t.PlaylistID, &t.Title, &t.Artist, &t.DurationMs,
			&t.ReleaseDate); err != nil {
			s.serverError(w, err)
			return
		}
		tracks = append(tracks, t)
	}
	if err := rows.Err(); err != nil {
		s.serverError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"playlist": pl,
		"tracks":   tracks,
	})
}
