// Command cpnpd serves the numeric-tensor-computation-pipeline API.
//
// Everything it needs is inside this tree: the seed file is ./testdata, the
// data store is created under ./data, and the listen address is a literal below
// rather than an allocation looked up in some table living somewhere else.
// Clone the repository, run this, and it works.
//
// Four environment variables override the defaults, each read at its use site:
//
//	CPNPD_ADDR          listen address              (default :8310)
//	CPNPD_DATA_PATH     data store file             (default ./data/cpnpd.dat)
//	CPNPD_SEED_PATH     local tensor-seed file      (default ./testdata/tensor_seed.ctf)
//	CPNPD_FLAGS         comma-separated flag list, read by the pipeline package
//
// The seed is loaded once, on the first boot against an empty data store. There
// is no runtime reload route: reseeding means stopping the process, clearing
// the data store file, and starting it again.
package main

import (
	"context"
	"errors"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"path/filepath"
	"strings"
	"syscall"
	"time"

	"cpnp/internal/pipeline"
	"cpnp/internal/ingest"
	"cpnp/internal/store"
)

func envOr(key, fallback string) string {
	if v := strings.TrimSpace(os.Getenv(key)); v != "" {
		return v
	}
	return fallback
}

func run(logger *slog.Logger) error {
	addr := envOr("CPNP_ADDR", ":8310")
	dataPath := envOr("CPNP_DATA_PATH", "./data/cpnpd.dat")
	seedPath := envOr("CPNP_SEED_PATH", "./testdata/tensor_seed.ctf")

	if err := os.MkdirAll(filepath.Dir(dataPath), 0o755); err != nil {
		return err
	}
	dataStore, err := store.Open(dataPath)
	if err != nil {
		return err
	}
	defer dataStore.Close()

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	loaded, counts, err := ingest.EnsureSeeded(ctx, dataStore, seedPath)
	if err != nil {
		return err
	}
	if loaded {
		logger.Info("seeded computation pipeline", "seed", seedPath, "tensors", counts.Tensors,
			"operations", counts.Operations, "results", counts.Results)
	}

	srv := &http.Server{
		Addr:              addr,
		Handler:           pipeline.NewServer(dataStore, logger).Handler(),
		ReadHeaderTimeout: 10 * time.Second,
	}
	go func() {
		<-ctx.Done()
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		_ = srv.Shutdown(shutdownCtx)
	}()

	logger.Info("listening", "addr", addr, "data", dataPath, "seed", seedPath)
	if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
		return err
	}
	logger.Info("stopped")
	return nil
}

func main() {
	logger := slog.New(slog.NewTextHandler(os.Stderr, &slog.HandlerOptions{
		Level: slog.LevelInfo,
	}))
	if err := run(logger); err != nil {
		logger.Error("cpnpd exited", "err", err)
		os.Exit(1)
	}
}
