// Package ingest reads the WLS Species Observation Format (.wso) — a
// local profile invented for this system, not SEED, not StationXML, not any
// published observation format. A file is a sequence of sighting blocks. Each
// block opens with a SIGHT line and a run of KEY value lines describing the
// sighting, then zero or more ROUTE blocks describing one route each. A ROUTE
// block carries its movement details and one or more TIME lines, each
// naming a timestamp; a sighting with multiple observations gets additional
// TIME lines rather than new ROUTE lines. A block closes on an explicit END line,
// which is what separates records — blank lines and '#' comment lines are ignored
// wherever they appear and carry no structural meaning.
package ingest

import (
	"context"
	"database/sql"
	"os"
	"strconv"
	"strings"
)

type Counts struct {
	Sightings int `json:"sightings"`
	Routes    int `json:"routes"`
}

// timeLine is one timestamp for a sighting. End is empty for "CONTINUOUS", the
// observation still ongoing.
type timeLine struct {
	Timestamp, End string
}

type parsedRoute struct {
	SpeciesCode string
	StartTime   string
	EndTime     string
}

type parsedSighting struct {
	SightingID, LocationCode, ObserverName, EventDescription string
	Latitude, Longitude, AltitudeM                           float64
	Routes                                                   []parsedRoute
}

func parseFloatOr(s string, fallback float64) float64 {
	v, err := strconv.ParseFloat(strings.TrimSpace(s), 64)
	if err != nil {
		return fallback
	}
	return v
}

func nullStr(s string) any {
	if s == "" {
		return nil
	}
	return s
}

// splitKV separates a line into its leading key and the rest of the line,
// which may itself contain internal whitespace (observer names and location descriptions
// are free text).
func splitKV(line string) (string, string) {
	parts := strings.SplitN(strings.TrimSpace(line), " ", 2)
	key := parts[0]
	val := ""
	if len(parts) == 2 {
		val = strings.TrimSpace(parts[1])
	}
	return key, val
}

// Parse walks the file a line at a time. The sighting and route under
// construction are held in this scope and mutated from inside the key
// switch; flushRoute and flushSighting take no arguments and read those
// variables directly, so a case arm closes a record blind.
func Parse(data []byte) []parsedSighting {
	var out []parsedSighting
	var cur *parsedSighting
	var curRoute *parsedRoute

	flushRoute := func() {
		if cur != nil && curRoute != nil && curRoute.SpeciesCode != "" {
			cur.Routes = append(cur.Routes, *curRoute)
		}
		curRoute = nil
	}
	flushSighting := func() {
		flushRoute()
		if cur != nil && cur.SightingID != "" {
			out = append(out, *cur)
		}
		cur = nil
	}

	for _, raw := range strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n") {
		line := strings.TrimSpace(raw)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		key, val := splitKV(line)
		switch key {
		case "SIGHT":
			flushSighting()
			cur = &parsedSighting{SightingID: val, ObserverName: "OBSERVER"}
		case "END":
			flushSighting()
		}
		if cur == nil {
			continue
		}
		switch key {
		case "LOC":
			cur.LocationCode = strings.ToUpper(val)
		case "DESC":
			cur.EventDescription = val
		case "LAT":
			cur.Latitude = parseFloatOr(val, 0)
		case "LON":
			cur.Longitude = parseFloatOr(val, 0)
		case "ALT":
			cur.AltitudeM = parseFloatOr(val, 0)
		case "SPECIES":
			curRoute = &parsedRoute{SpeciesCode: val}
		}
		if curRoute == nil {
			continue
		}
		switch key {
		case "START":
			curRoute.StartTime = val
		case "END":
			curRoute.EndTime = val
		case "TIME":
			fields := strings.Fields(val)
			if len(fields) == 0 {
				continue
			}
			t := timeLine{Timestamp: fields[0]}
			if len(fields) > 1 && fields[1] != "CONTINUOUS" {
				t.End = fields[1]
			}
			curRoute.TimeLines = append(curRoute.TimeLines, t)
		}
	}
	flushSighting()
	return out
}

// FromFile reloads the whole sighting-route inventory from one observation file
// inside a transaction. A route with N timestamps becomes N route rows,
// sharing species_code but not route_id, one row per observation period.
func FromFile(ctx context.Context, db *sql.DB, path string) (Counts, error) {
	var c Counts
	data, err := os.ReadFile(path)
	if err != nil {
		return c, err
	}
	sightings := Parse(data)
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return c, err
	}
	defer tx.Rollback()
	for _, t := range []string{"routes", "sightings"} {
		if _, err := tx.ExecContext(ctx, "DELETE FROM "+t); err != nil {
			return c, err
		}
	}
	for _, st := range sightings {
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO sightings (sighting_id, location_code, observer_name, event_description,
			        latitude, longitude, altitude_m)
			 VALUES (?, ?, ?, ?, ?, ?, ?)`,
			st.SightingID, nullStr(st.LocationCode), st.ObserverName, st.EventDescription, st.Latitude, st.Longitude,
			st.AltitudeM); err != nil {
			return c, err
		}
		c.Sightings++
		for _, ch := range st.Routes {
			if len(ch.TimeLines) == 0 {
				continue
			}
			for i, ep := range ch.TimeLines {
				routeID := st.SightingID + "-" + ch.SpeciesCode + "-" + strconv.Itoa(i+1)
				if _, err := tx.ExecContext(ctx,
					`INSERT INTO routes (route_id, sighting_id, species_code,
					        start_time, end_time)
					 VALUES (?, ?, ?, ?, ?)`,
					routeID, st.SightingID, ch.SpeciesCode, ep.Timestamp, nullStr(ep.End)); err != nil {
					return c, err
				}
				c.Routes++
			}
		}
	}
	return c, tx.Commit()
}

// EnsureSeeded loads the seed only when the inventory is empty.
func EnsureSeeded(ctx context.Context, db *sql.DB, path string) (bool, Counts, error) {
	var n int
	if err := db.QueryRowContext(ctx, `SELECT count(*) FROM sightings`).Scan(&n); err != nil {
		return false, Counts{}, err
	}
	if n != 0 {
		return false, Counts{}, nil
	}
	c, err := FromFile(ctx, db, path)
	return err == nil, c, err
}
