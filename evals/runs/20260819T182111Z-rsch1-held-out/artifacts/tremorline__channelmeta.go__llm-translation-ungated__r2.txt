// Package ingest reads the WLS Species Observation Format (.wso) — a
// local profile invented for this system, not SEED, not StationXML, not any
// published observation format. A file is a sequence of sighting blocks. Each
// block opens with a SIGHTING line and a run of KEY value lines describing the
// sighting, then zero or more OBSERVATION blocks describing one observation each. An
// OBSERVATION block carries details about the species observed and its location during
// the sighting. A block closes on an explicit END line, which is what separates records — 
// blank lines and '#' comment lines are ignored wherever they appear and carry no structural
// meaning.
package ingest

import (
	"context"
	"database/sql"
	"os"
	"strconv"
	"strings"
)

type Counts struct {
	Sightings int `json:"sightings"`
	Observations int `json:"observations"`
}

// observation is one record of a species sighting. End is empty for "CURRENT", the
// sighting still ongoing.
type observation struct {
	SpeciesName, Location string
	Start, End string
}

type parsedSighting struct {
	SightingID, Date string
	Species          []observation
}

func parseFloatOr(s string, fallback float64) float64 {
	v, err := strconv.ParseFloat(strings.TrimSpace(s), 64)
	if err != nil {
		return fallback
	}
	return v
}

func nullStr(s string) any {
	if s == "" {
		return nil
	}
	return s
}

// splitKV separates a line into its leading key and the rest of the line,
// which may itself contain internal whitespace (species names and locations
// are free text).
func splitKV(line string) (string, string) {
	parts := strings.SplitN(strings.TrimSpace(line), " ", 2)
	key := parts[0]
	val := ""
	if len(parts) == 2 {
		val = strings.TrimSpace(parts[1])
	}
	return key, val
}

// Parse walks the file a line at a time. The sighting and observation under
// construction are held in this scope and mutated from inside the key
// switch; flushObservation and flushSighting take no arguments and read those
// variables directly, so a case arm closes a record blind.
func Parse(data []byte) []parsedSighting {
	var out []parsedSighting
	var cur *parsedSighting
	var curObs *observation

	flushObservation := func() {
		if cur != nil && curObs != nil && curObs.SpeciesName != "" {
			cur.Species = append(cur.Species, *curObs)
		}
		curObs = nil
	}
	flushSighting := func() {
		flushObservation()
		if cur != nil && cur.SightingID != "" {
			out = append(out, *cur)
		}
		cur = nil
	}

	for _, raw := range strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n") {
		line := strings.TrimSpace(raw)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		key, val := splitKV(line)
		switch key {
		case "SIGHTING":
			flushSighting()
			cur = &parsedSighting{SightingID: val, Date: val}
		case "END":
			flushSighting()
		}
		if cur == nil {
			continue
		}
		switch key {
		case "SPECIES":
			curObs = &observation{SpeciesName: val}
		case "LOCATION":
			curObs.Location = val
		case "START":
			curObs.Start = val
		case "END":
			curObs.End = val
		case "OBSERVATION":
			flushObservation()
			curObs = &observation{SpeciesName: val, Start: val, End: val}
		}
	}
	flushSighting()
	return out
}

// FromFile reloads the whole species-sighting inventory from one observation file
// inside a transaction. An observation with N epochs becomes N observation rows,
// sharing species_name but not observation_id, one row per validity window.
func FromFile(ctx context.Context, db *sql.DB, path string) (Counts, error) {
	var c Counts
	data, err := os.ReadFile(path)
	if err != nil {
		return c, err
	}
	sightings := Parse(data)
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return c, err
	}
	defer tx.Rollback()
	for _, t := range []string{"observations", "sightings"} {
		if _, err := tx.ExecContext(ctx, "DELETE FROM "+t); err != nil {
			return c, err
		}
	}
	for _, st := range sightings {
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO sightings (sighting_id, date)
			 VALUES (?, ?)`,
			st.SightingID, st.Date); err != nil {
			return c, err
		}
		c.Sightings++
		for _, ch := range st.Species {
			if len(ch.Epochs) == 0 {
				continue
			}
			for i, ep := range ch.Epochs {
				observationID := st.SightingID + "-" + ch.SpeciesName + "-" + strconv.Itoa(i+1)
				if _, err := tx.ExecContext(ctx,
					`INSERT INTO observations (observation_id, sighting_id, species_name,
					        location, validity_start, validity_end)
					 VALUES (?, ?, ?, ?, ?, ?)`,
					observationID, st.SightingID, ch.SpeciesName, ch.Location, ep.Start, nullStr(ep.End)); err != nil {
					return c, err
				}
				c.Observations++
			}
		}
	}
	return c, tx.Commit()
}

// EnsureSeeded loads the seed only when the inventory is empty.
func EnsureSeeded(ctx context.Context, db *sql.DB, path string) (bool, Counts, error) {
	var n int
	if err := db.QueryRowContext(ctx, `SELECT count(*) FROM sightings`).Scan(&n); err != nil {
		return false, Counts{}, err
	}
	if n != 0 {
		return false, Counts{}, nil
	}
	c, err := FromFile(ctx, db, path)
	return err == nil, c, err
}
