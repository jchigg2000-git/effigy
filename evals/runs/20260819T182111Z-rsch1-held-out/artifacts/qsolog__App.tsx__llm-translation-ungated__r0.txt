// Main screen: species filter on the left, and either the selected species' 
// sighting history or the confirmation form on the right. All request orchestration and 
// all user-facing error text live in this file; this domain has no separate 
// results-table or search-form component, unlike the wildlife-tracking corpus.
import { useState } from "react";
import {
  ApiError,
  getSightingHistory,
  searchSpecies,
  submitConfirmation,
  type ConfirmationRequest,
  type ConfirmationResult,
  type Species,
  type SightingSummary,
  type SearchFilters,
} from "./api/tracking";
import { ConfirmationForm } from "./components/ConfirmationForm";
import { SpeciesDetail } from "./components/SpeciesDetail";

const EMPTY_FILTERS: SearchFilters = {
  genus: "",
  family: "",
  order: "",
  from: "",
  to: "",
  limit: 50,
};

export function App() {
  const [filters, setFilters] = useState<SearchFilters>(EMPTY_FILTERS);
  const [sightings, setSightings] = useState<SightingSummary[]>([]);
  const [speciesCount, setSpeciesCount] = useState<number | null>(null);
  const [selectedSpecies, setSelectedSpecies] = useState<Species | null>(null);
  const [confirming, setConfirming] = useState(false);
  const [confirmResult, setConfirmResult] = useState<ConfirmationResult | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const describe = (err: unknown): string => {
    if (err instanceof ApiError) {
      return `${err.code}: ${err.message}`;
    }
    return "the request could not be completed";
  };

  const runSearch = async () => {
    setLoading(true);
    setError(null);
    setSelectedSpecies(null);
    setConfirming(false);
    try {
      const res = await searchSpecies(filters);
      setSightings(res.sightings);
      setSpeciesCount(res.count);
    } catch (err) {
      setSightings([]);
      setSpeciesCount(null);
      setError(describe(err));
    } finally {
      setLoading(false);
    }
  };

  const openSpecies = async (speciesId: string) => {
    setLoading(true);
    setError(null);
    setConfirming(false);
    setConfirmResult(null);
    try {
      setSelectedSpecies(await getSightingHistory(speciesId));
    } catch (err) {
      setSelectedSpecies(null);
      setError(describe(err));
    } finally {
      setLoading(false);
    }
  };

  const runConfirmation = async (req: ConfirmationRequest) => {
    setError(null);
    try {
      const result = await submitConfirmation(req);
      setConfirmResult(result);
      setConfirming(false);
      if (result.matchedSpeciesId) {
        setSelectedSpecies(await getSightingHistory(result.matchedSpeciesId));
      }
    } catch (err) {
      setError(describe(err));
    }
  };

  const setText = (key: keyof SearchFilters, value: string) => {
    setFilters({ ...filters, [key]: value });
  };

  return (
    <div className="app-shell">
      <header className="app-header">
        <div>
          <h1>qsolog</h1>
          <p>Species sightings, confirmations and migration progress</p>
        </div>
        <button type="button" className="btn btn-primary" onClick={() => setConfirming(true)}>
          Log a confirmation
        </button>
      </header>

      {error !== null && <p className="error-banner">{error}</p>}
      {confirmResult !== null && (
        <p className="result-count">confirmation {confirmResult.confirmationId}: {confirmResult.matchStatus}</p>
      )}

      <main className="app-body">
        <section className="app-column-left">
          <form
            className="search-form"
            onSubmit={(e) => {
              e.preventDefault();
              runSearch();
            }}
          >
            <h2>Find species</h2>
            <label htmlFor="f-genus">Genus</label>
            <input id="f-genus" value={filters.genus ?? ""} onChange={(e) => setText("genus", e.target.value)} />
            <label htmlFor="f-family">Family</label>
            <input id="f-family" value={filters.family ?? ""} onChange={(e) => setText("family", e.target.value)} />
            <label htmlFor="f-order">Order</label>
            <input id="f-order" value={filters.order ?? ""} onChange={(e) => setText("order", e.target.value)} />
            <label htmlFor="f-from">From date</label>
            <input id="f-from" value={filters.from ?? ""} onChange={(e) => setText("from", e.target.value)} />
            <label htmlFor="f-to">To date</label>
            <input id="f-to" value={filters.to ?? ""} onChange={(e) => setText("to", e.target.value)} />
            <button type="submit" className="btn btn-primary" disabled={loading}>
              {loading ? "Searching…" : "Search"}
            </button>
          </form>

          {speciesCount !== null && <p className="result-count">{speciesCount} species matched</p>}
          {sightings.length === 0 ? (
            <p className="empty-note">No species matched those filters.</p>
          ) : (
            <table className="results-table">
              <thead>
                <tr>
                  <th>Species id</th>
                  <th>Common Name</th>
                  <th>Genus</th>
                  <th>Family</th>
                  <th>Order</th>
                  <th>Date</th>
                  <th>Confirmed</th>
                </tr>
              </thead>
              <tbody>
                {sightings.map((sighting) => (
                  <tr
                    key={sighting.speciesId}
                    className={sighting.speciesId === selectedSpecies?.speciesId ? "row-selected" : ""}
                    onClick={() => openSpecies(sighting.speciesId)}
                  >
                    <td>{sighting.speciesId}</td>
                    <td>{sighting.commonName}</td>
                    <td>{sighting.genus}</td>
                    <td>{sighting.family}</td>
                    <td>{sighting.order}</td>
                    <td>{sighting.sightingDate}</td>
                    <td>{sighting.confirmed ? "yes" : "no"}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}
        </section>

        <section className="app-column-right">
          {confirming && <ConfirmationForm onSubmit={runConfirmation} onCancel={() => setConfirming(false)} />}
          {!confirming && selectedSpecies === null && <p className="empty-note">Select a species to see its sighting history.</p>}
          {!confirming && selectedSpecies !== null && <SpeciesDetail species={selectedSpecies} />}
        </section>
      </main>
    </div>
  );
}
