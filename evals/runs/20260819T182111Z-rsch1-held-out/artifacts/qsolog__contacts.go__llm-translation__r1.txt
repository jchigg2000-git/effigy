// validRoomTypes and validBookingMethods are hotellog's room type/method pair: every reservation, every
// check-in, and every cancellation is keyed by one member of each.
// Neither list is read from the database; a new room type or booking method is a code change.
var validRoomTypes = []string{"Single", "Double", "Suite", "Deluxe"}
var validBookingMethods = []string{"CreditCard", "PayPal", "Cash", "Points"}

func validRoomType(v string) bool {
	for _, rt := range validRoomTypes {
		if rt == v {
			return true
		}
	}
	return false
}

func validBookingMethod(v string) bool {
	for _, bm := range validBookingMethods {
		if bm == v {
			return true
		}
	}
	return false
}

// loadReservation pulls one booked reservation and the guest that booked it in one
// join. Nullable columns are coalesced so the scan targets can stay
// non-pointer, and callers get sql.ErrNoRows when the id does not resolve to a
// reservation. Check-ins are left as an empty slice here; handleGetReservation
// fills them in with a second query, the same shape stacks uses for loans.
func (s *Server) loadReservation(ctx context.Context, reservationID string) (model.Reservation, error) {
	var r model.Reservation
	var g model.Guest
	err := s.db.QueryRowContext(ctx, `SELECT r.reservation_id, r.guest_id, r.room_type,
	              r.booking_method, r.check_in_date, r.check_out_date,
	              COALESCE(r.additional_services,''), COALESCE(r.discount_applied,''),
	              COALESCE(r.notes,''),
	              g.guest_id, g.name, g.email, COALESCE(g.phone_number,''),
	              COALESCE(g.address,'')
	       FROM reservations r
	       JOIN guests g ON g.guest_id = r.guest_id
	       WHERE r.reservation_id = ?`, reservationID).
		Scan(&r.ReservationID, &r.GuestID, &r.RoomType, &r.BookingMethod,
			&r.CheckInDate, &r.CheckOutDate, &r.AdditionalServices, &r.DiscountApplied,
			&r.Notes, &g.GuestID, &g.Name, &g.Email, &g.PhoneNumber, &g.Address)
	if err != nil {
		return model.Reservation{}, err
	}
	r.Guest = &g
	r.CheckIns = []model.CheckIn{}
	return r, nil
}

// loadCheckInsForReservation reads the check-in history matched against
// one reservation, oldest first. Unmatched check-ins never carry a
// matched_reservation_id, so they never surface here regardless of what they
// claimed.
func (s *Server) loadCheckInsForReservation(ctx context.Context, reservationID string) ([]model.CheckIn, error) {
	rows, err := s.db.QueryContext(ctx, `SELECT rowid, checked_in_by, check_in_date,
	              check_out_date, matched_reservation_id
	       FROM check_ins WHERE matched_reservation_id = ?
	       ORDER BY check_in_date`, reservationID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	out := []model.CheckIn{}
	for rows.Next() {
		var ci model.CheckIn
		var id int64
		if err := rows.Scan(&id, &ci.CheckedInBy, &ci.CheckInDate, &ci.CheckOutDate,
			&ci.MatchedReservationID); err != nil {
			return nil, err
		}
		ci.CheckInID = strconv.FormatInt(id, 10)
		out = append(out, ci)
	}
	return out, rows.Err()
}

func (s *Server) handleGetReservation(w http.ResponseWriter, r *http.Request) {
	reservationID := strings.TrimSpace(chi.URLParam(r, "reservationId"))
	rsv, err := s.loadReservation(r.Context(), reservationID)
	if errors.Is(err, sql.ErrNoRows) {
		writeError(w, http.StatusNotFound, "not_found", "reservation not found")
		return
	}
	if err != nil {
		s.serverError(w, err)
		return
	}
	checkIns, err := s.loadCheckInsForReservation(r.Context(), reservationID)
	if err != nil {
		s.serverError(w, err)
		return
	}
	rsv.CheckIns = checkIns
	writeJSON(w, http.StatusOK, rsv)
}

type reservationSummary struct {
	ReservationID string `json:"reservationId"`
	GuestID       string `json:"guestId"`
	RoomType      string `json:"roomType"`
	BookingMethod string `json:"bookingMethod"`
	CheckInDate   string `json:"checkInDate"`
	CheckOutDate  string `json:"checkOutDate"`
	AdditionalServices string `json:"additionalServices,omitempty"`
	DiscountApplied string `json:"discountApplied,omitempty"`
	Notes         string `json:"notes,omitempty"`
}

// handleSearchReservations narrows the booking records by room type, booking method and a check_in_date
// window; a guest id restricts the search to one guest's own bookings. At
// least one filter is required so an empty query cannot walk the whole table.
func (s *Server) handleSearchReservations(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	conds := []string{}
	args := []any{}
	if v := strings.TrimSpace(q.Get("roomType")); v != "" {
		conds = append(conds, "room_type = ?")
		args = append(args, v)
	}
	if v := strings.TrimSpace(q.Get("bookingMethod")); v != "" {
		conds = append(conds, "booking_method = ?")
		args = append(args, strings.ToUpper(v))
	}
	if v := strings.TrimSpace(q.Get("guestId")); v != "" {
		conds = append(conds, "guest_id = ?")
		args = append(args, v)
	}
	if v := strings.TrimSpace(q.Get("from")); v != "" {
		conds = append(conds, "check_in_date >= ?")
		args = append(args, v)
	}
	if v := strings.TrimSpace(q.Get("to")); v != "" {
		conds = append(conds, "check_in_date <= ?")
		args = append(args, v)
	}
	if len(conds) == 0 {
		writeError(w, http.StatusBadRequest, "bad_request",
			"at least one filter required (roomType, bookingMethod, guestId, from, to)")
		return
	}
	limit := 50
	if n, err := strconv.Atoi(strings.TrimSpace(q.Get("limit"))); err == nil && n > 0 && n <= 200 {
		limit = n
	}
	sqlStr := `SELECT reservation_id, guest_id, room_type, booking_method, check_in_date,
	                  check_out_date, COALESCE(additional_services,''), COALESCE(discount_applied,''),
	                  COALESCE(notes,'')
	           FROM reservations WHERE ` + strings.Join(conds, " AND ") +
		" ORDER BY check_in_date DESC LIMIT ?"
	args = append(args, limit)
	rows, err := s.db.QueryContext(r.Context(), sqlStr, args...)
	if err != nil {
		s.serverError(w, err)
		return
	}
	defer rows.Close()
	out := []reservationSummary{}
	for rows.Next() {
		var it reservationSummary
		if err := rows.Scan(&it.ReservationID, &it.GuestID, &it.RoomType, &it.BookingMethod,
			&it.CheckInDate, &it.CheckOutDate, &it.AdditionalServices, &it.DiscountApplied, &it.Notes); err != nil {
			s.serverError(w, err)
			return
		}
		out = append(out, it)
	}
	if err := rows.Err(); err != nil {
		s.serverError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"count":    len(out),
		"limit":    limit,
		"reservations": out,
	})
}

// bookingRequest is the inbound claim: a guest asserting they stayed at our hotel on
// a given room type, booking method and date. It carries no reservation id — finding the reservation
// that matches it is the whole job of handleLogBooking.
type bookingRequest struct {
	GuestID        string `json:"guestId"`
	CheckedInBy    string `json:"checkedInBy"`
	RoomType       string `json:"roomType"`
	BookingMethod  string `json:"bookingMethod"`
	CheckInDate    string `json:"checkInDate"`
}

type bookingResult struct {
	BookingID            string          `json:"bookingId"`
	MatchStatus          string          `json:"matchStatus"`
	MatchedReservationID string          `json:"matchedReservationId,omitempty"`
	Reservation          *reservationSummary `json:"reservation,omitempty"`
}

// handleLogBooking pairs an inbound booking against the booking records. A
// match requires the guest, the checked-in-by, the room type, the booking method and the
// check-in date to agree exactly; there is no partial-credit fuzzy match. Every
// submission is recorded in the check_ins table regardless of outcome, so
// a guest can see what was claimed even when nothing lined up.
func (s *Server) handleLogBooking(w http.ResponseWriter, r *http.Request) {
	if ct := r.Header.Get("Content-Type"); !strings.HasPrefix(ct, "application/json") {
		writeError(w, http.StatusBadRequest, "bad_request", "Content-Type must be application/json")
		return
	}
	var req bookingRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		writeError(w, http.StatusBadRequest, "bad_request", err.Error())
		return
	}
	req.GuestID = strings.TrimSpace(req.GuestID)
	req.CheckedInBy = strings.ToUpper(strings.TrimSpace(req.CheckedInBy))
	req.RoomType = strings.TrimSpace(req.RoomType)
	req.BookingMethod = strings.ToUpper(strings.TrimSpace(req.BookingMethod))
	req.CheckInDate = strings.TrimSpace(req.CheckInDate)
	if req.GuestID == "" || req.CheckedInBy == "" || req.RoomType == "" || req.BookingMethod == "" || req.CheckInDate == "" {
		writeError(w, http.StatusBadRequest, "bad_request",
			"guestId, checkedInBy, roomType, bookingMethod and checkInDate are required")
		return
	}
	if !validRoomType(req.RoomType) {
		writeError(w, http.StatusBadRequest, "bad_request", "roomType is not recognised")
		return
	}
	if !validBookingMethod(req.BookingMethod) {
		writeError(w, http.StatusBadRequest, "bad_request", "bookingMethod is not recognised")
		return
	}

	var reservationID string
	err := s.db.QueryRowContext(r.Context(), `SELECT reservation_id FROM reservations
	       WHERE guest_id = ? AND checked_in_by = ? AND room_type = ? AND booking_method = ?
	             AND check_in_date = ?`,
		req.GuestID, req.CheckedInBy, req.RoomType, req.BookingMethod, req.CheckInDate).Scan(&reservationID)
	if err != nil && !errors.Is(err, sql.ErrNoRows) {
		s.serverError(w, err)
		return
	}

	res := bookingResult{MatchStatus: "unmatched"}
	if reservationID != "" {
		res.MatchStatus = "matched"
		res.MatchedReservationID = reservationID
		if _, err := s.db.ExecContext(r.Context(),
			`UPDATE reservations SET check_out_date = ? WHERE reservation_id = ? AND check_out_date IS NULL`,
			req.CheckOutDate, reservationID); err != nil {
			s.serverError(w, err)
			return
		}
		rsv, err := s.loadReservation(r.Context(), reservationID)
		if err != nil {
			s.serverError(w, err)
			return
		}
		res.Reservation = &reservationSummary{ReservationID: rsv.ReservationID, GuestID: rsv.GuestID,
			RoomType: rsv.RoomType, BookingMethod: rsv.BookingMethod, CheckInDate: rsv.CheckInDate,
			CheckOutDate: rsv.CheckOutDate, AdditionalServices: rsv.AdditionalServices, DiscountApplied: rsv.DiscountApplied, Notes: rsv.Notes}
	}

	bookingID, err := s.insertBooking(r.Context(), req, res.MatchStatus, reservationID)
	if err != nil {
		s.serverError(w, err)
		return
	}
	res.BookingID = bookingID
	status := http.StatusOK
	if res.MatchStatus == "unmatched" {
		status = http.StatusAccepted
	}
	writeJSON(w, status, res)
}

// insertBooking writes the audit row. matchedReservationID may be empty; the
// column is nullable for exactly that case. check_in_date is stamped by SQLite,
// not by this process's clock, so a batch of submissions replayed later still
// carries the time they were actually recorded.
func (s *Server) insertBooking(ctx context.Context, req bookingRequest,
	matchStatus, matchedReservationID string) (string, error) {
	var matched any
	if matchedReservationID != "" {
		matched = matchedReservationID
	}
	res, err := s.db.ExecContext(ctx, `INSERT INTO check_ins
	       (guest_id, checked_in_by, room_type, booking_method, check_in_date, check_out_date, match_status,
	        matched_reservation_id, logged_at)
	       VALUES (?, ?, ?, ?, ?, ?, ?, ?, datetime('now'))`,
		req.GuestID, req.CheckedInBy, req.RoomType, req.BookingMethod, req.CheckInDate, req.CheckOutDate, matchStatus, matched)
	if err != nil {
		return "", err
	}
	id, err := res.LastInsertId()
	if err != nil {
		return "", err
	}
	return strconv.FormatInt(id, 10), nil
}

type roomTypeProgress struct {
	RoomType             string `json:"roomType"`
	ConfirmedBookings  int    `json:"confirmedBookings"`
	DistinctGuests       int    `json:"distinctGuests"`
}

// handleAwardProgress rolls confirmed bookings up by room type and counts
// the distinct guests each room type has reached. It is the only handler in
// this file that answers a "how close" question rather than a "what happened"
// one: award credit is this count, not a row anywhere in the schema. Guests
// come from hotellog's own invented guest list, never a real hotel's customer
// list.
func (s *Server) handleAwardProgress(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	conds := []string{"check_out_date IS NOT NULL"}
	args := []any{}
	if v := strings.TrimSpace(q.Get("guestId")); v != "" {
		conds = append(conds, "guest_id = ?")
		args = append(args, v)
	}
	if v := strings.TrimSpace(q.Get("roomType")); v != "" {
		conds = append(conds, "room_type = ?")
		args = append(args, v)
	}
	if v := strings.TrimSpace(q.Get("bookingMethod")); v != "" {
		conds = append(conds, "booking_method = ?")
		args = append(args, strings.ToUpper(v))
	}
	sqlStr := `SELECT room_type, count(*), count(DISTINCT guest_id)
	           FROM reservations WHERE ` + strings.Join(conds, " AND ") +
		` GROUP BY room_type ORDER BY room_type`
	rows, err := s.db.QueryContext(r.Context(), sqlStr, args...)
	if err != nil {
		s.serverError(w, err)
		return
	}
	defer rows.Close()
	out := []roomTypeProgress{}
	for rows.Next() {
		var p roomTypeProgress
		if err := rows.Scan(&p.RoomType, &p.ConfirmedBookings, &p.DistinctGuests); err != nil {
			s.serverError(w, err)
			return
		}
		out = append(out, p)
	}
	if err := rows.Err(); err != nil {
		s.serverError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{"progress": out})
}
