// Package process reads the NUMF (Numeric Feed) fixed-width feed
// into the data processing system. NUMF is a local profile invented for this
// corpus item — it names no real system vendor and follows no published
// exchange standard. Layout below; columns are 0-indexed for Go slicing.
//
//	Header (one line, type 'H'):
//	  [0:1]   record type, always "H"
//	  [1:9]   file timestamp, YYYYMMDDHHmmss
//	  [9:15]  task code, e.g. "TK0007"
//	  [15:23] batch code, e.g. "20260701000000"
//	  [23:40] reserved, space-filled
//
//	Detail (one line per entry, type 'D'):
//	  [0:1]   record type, always "D"
//	  [1:11]  entry id, e.g. "ENT-0000001"
//	  [11:21] tensor id, e.g. "TSR-000001"
//	  [21:29] entry timestamp, YYYYMMDDHHmmss
//	  [29:39] entry value, zero-padded floating-point units
//	  [39:40] entry type code: S scalar, V vector, M matrix
//	  [40:41] precision flag: Y or N
//	  [41:60] reserved, space-filled
//
//	Trailer (one line, type 'T'):
//	  [0:1]  record type, always "T"
//	  [1:9]  detail record count, zero-padded
//	  [9:20] reserved, space-filled
//
// Exactly one header opens the file and exactly one trailer closes it; the
// trailer's declared count is checked against the detail lines actually
// parsed, and a mismatch is a parse error rather than a warning a caller
// could shrug off.
package process

import (
	"context"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type Counts struct {
	Entries     int `json:"entries"`
	Tensors     int `json:"tensors"`
	Scalars     int `json:"scalars"`
	Vectors     int `json:"vectors"`
	Matrices    int `json:"matrices"`
	BatchCodes  int `json:"batchCodes"`
}

type parsedEntry struct {
	EntryID, TensorID, EntryTimestamp string
	EntryValue                        float64
	EntryType                         string
	PrecisionFlag                     bool
}

type parsedFile struct {
	FileTimestamp, TaskCode, BatchCode string
	Entries                           []parsedEntry
}

var entryTypeNames = map[byte]string{'S': "SCALAR", 'V': "VECTOR", 'M': "MATRIX"}

func field(line string, start, end int) string {
	if end > len(line) {
		end = len(line)
	}
	if start >= end {
		return ""
	}
	return strings.TrimSpace(line[start:end])
}

// Parse walks the file a line at a time. Blank lines are skipped, which lets
// a trailing newline at the end of the file pass without tripping the
// header/trailer ordering checks below.
func Parse(data []byte) (*parsedFile, error) {
	pf := &parsedFile{}
	sawHeader := false
	sawTrailer := false
	declaredCount := 0

	lines := strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n")
	for _, raw := range lines {
		if strings.TrimSpace(raw) == "" {
			continue
		}
		switch raw[0] {
		case 'H':
			if sawHeader {
				return nil, fmt.Errorf("numf: duplicate header line")
			}
			sawHeader = true
			pf.FileTimestamp = field(raw, 1, 9)
			pf.TaskCode = field(raw, 9, 15)
			pf.BatchCode = field(raw, 15, 23)
		case 'D':
			if !sawHeader {
				return nil, fmt.Errorf("numf: detail line before header")
			}
			if sawTrailer {
				return nil, fmt.Errorf("numf: detail line after trailer")
			}
			value, err := strconv.ParseFloat(field(raw, 29, 39), 64)
			if err != nil {
				return nil, fmt.Errorf("numf: bad entry value: %w", err)
			}
			var typeCode byte
			if tc := field(raw, 39, 40); tc != "" {
				typeCode = tc[0]
			}
			entryType, ok := entryTypeNames[typeCode]
			if !ok {
				return nil, fmt.Errorf("numf: unknown entry type code %q", string(typeCode))
			}
			pf.Entries = append(pf.Entries, parsedEntry{
				EntryID: field(raw, 1, 11),
				TensorID: field(raw, 11, 21),
				EntryTimestamp: field(raw, 21, 29),
				EntryValue:      value,
				EntryType:       entryType,
				PrecisionFlag:  field(raw, 40, 41) == "Y",
			})
		case 'T':
			if sawTrailer {
				return nil, fmt.Errorf("numf: duplicate trailer line")
			}
			sawTrailer = true
			n, err := strconv.Atoi(field(raw, 1, 9))
			if err != nil {
				return nil, fmt.Errorf("numf: bad trailer count: %w", err)
			}
			declaredCount = n
		default:
			return nil, fmt.Errorf("numf: unrecognised record type %q", string(raw[0]))
		}
	}
	if !sawHeader {
		return nil, fmt.Errorf("numf: missing header line")
	}
	if !sawTrailer {
		return nil, fmt.Errorf("numf: missing trailer line")
	}
	if declaredCount != len(pf.Entries) {
		return nil, fmt.Errorf("numf: trailer declared %d detail records, parsed %d",
			declaredCount, len(pf.Entries))
	}
	return pf, nil
}

func nullStr(s string) any {
	if s == "" {
		return nil
	}
	return s
}

// FromFile reloads the data processing system from one NUMF feed inside a
// transaction. Entries and tensors are upserted since the same ones
// recur across every batch's feed; entries are always inserted fresh because an entry id is unique to its file and line.
func FromFile(ctx context.Context, system *System, path string) (Counts, error) {
	var c Counts
	data, err := os.ReadFile(path)
	if err != nil {
		return c, err
	}
	pf, err := Parse(data)
	if err != nil {
		return c, err
	}
	tx, err := system.BeginTransaction(ctx, nil)
	if err != nil {
		return c, err
	}
	defer tx.Rollback()

	if pf.BatchCode != "" {
		if _, err := tx.ExecuteCommand(ctx,
			`INSERT OR IGNORE INTO batches (batch_code, status, task_code)
			 VALUES (?, 'ACTIVE', ?)`, pf.BatchCode, pf.TaskCode); err != nil {
			return c, err
		}
		c.BatchCodes++
	}

	for i, ent := range pf.Entries {
		if ent.EntryID == "" || ent.TensorID == "" {
			continue
		}
		res, err := tx.ExecuteCommand(ctx,
			`INSERT OR IGNORE INTO entries (entry_id, tensor_id, task_code) VALUES (?, ?, ?)`,
			ent.EntryID, pf.TaskCode)
		if err != nil {
			return c, err
		}
		if n, _ := res.RowsAffected(); n > 0 {
			c.Entries++
		}
		res, err = tx.ExecuteCommand(ctx,
			`INSERT OR IGNORE INTO tensors (tensor_id, serial_number, entry_id)
			 VALUES (?, ?, ?)`, ent.TensorID, fmt.Sprintf("SN%08d", i+1), ent.EntryID)
		if err != nil {
			return c, err
		}
		if n, _ := res.RowsAffected(); n > 0 {
			c.Tensors++
		}
		entryID := fmt.Sprintf("ENT-%s-%04d", pf.BatchCode, i+1)
		if _, err := tx.ExecuteCommand(ctx,
			`INSERT INTO processed_entries (entry_id, tensor_id, entry_timestamp, task_code, batch_code,
			        entry_type, entry_value, precision_flag)
			 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
			entryID, ent.TensorID, ent.EntryTimestamp, pf.TaskCode, nullStr(pf.BatchCode),
			ent.EntryType, ent.EntryValue, ent.PrecisionFlag); err != nil {
			return c, err
		}
		c.Scalars++
		if ent.EntryType == "VECTOR" {
			c.Vectors++
		} else if ent.EntryType == "MATRIX" {
			c.Matrices++
		}
	}
	return c, tx.Commit()
}

// EnsureSeeded loads the seed only when the data processing system is empty.
func EnsureSeeded(ctx context.Context, system *System, path string) (bool, Counts, error) {
	var n int
	if err := system.QuerySingle(ctx, `SELECT count(*) FROM processed_entries`).Scan(&n); err != nil {
		return false, Counts{}, err
	}
	if n != 0 {
		return false, Counts{}, nil
	}
	c, err := FromFile(ctx, system, path)
	return err == nil, c, err
}
