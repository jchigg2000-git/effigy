// Package ingest reads the WLS Species Observation Format (.wso) — a
// local profile invented for this system, not SEED, not StationXML, not any
// published exchange format. A file is a sequence of observation blocks. Each
// block opens with an OBS line and a run of KEY value lines describing the
// observation, then zero or more SPEC blocks describing one species each. A SPEC
// block carries its sighting details and one or more ROUTE lines, each
// naming a migration route; a species observed multiple times gets a
// second SPEC block rather than a second ROUTE line. A block closes on an
// explicit END line, which is what separates records — blank lines and '#'
// comment lines are ignored wherever they appear and carry no structural
// meaning.
package ingest

import (
	"context"
	"database/sql"
	"os"
	"strconv"
	"strings"
)

type Counts struct {
	Species  int `json:"species"`
	Sightings int `json:"sightings"`
	Routes   int `json:"routes"`
}

// route is one migration route for a species. End is empty for "CONTINUOUS", the
// route still active.
type Route struct {
	Start, End string
}

type ParsedSighting struct {
	SpeciesCode string
	Latitude    float64
	Longitude   float64
	Route       []Route
}

type ParsedObservation struct {
	ObservationID, LocationCode, Date string
	Species                           []ParsedSighting
}

func parseFloatOr(s string, fallback float64) float64 {
	v, err := strconv.ParseFloat(strings.TrimSpace(s), 64)
	if err != nil {
		return fallback
	}
	return v
}

func nullStr(s string) any {
	if s == "" {
		return nil
	}
	return s
}

// splitKV separates a line into its leading key and the rest of the line,
// which may itself contain internal whitespace (location and date descriptions
// are free text).
func splitKV(line string) (string, string) {
	parts := strings.SplitN(strings.TrimSpace(line), " ", 2)
	key := parts[0]
	val := ""
	if len(parts) == 2 {
		val = strings.TrimSpace(parts[1])
	}
	return key, val
}

// Parse walks the file a line at a time. The observation and sighting under
// construction are held in this scope and mutated from inside the key
// switch; flushSighting and flushObservation take no arguments and read those
// variables directly, so a case arm closes a record blind.
func Parse(data []byte) []ParsedObservation {
	var out []ParsedObservation
	var cur *ParsedObservation
	var curSpec *ParsedSighting

	flushSighting := func() {
		if cur != nil && curSpec != nil && curSpec.SpeciesCode != "" {
			cur.Sightings = append(cur.Sightings, *curSpec)
		}
		curSpec = nil
	}
	flushObservation := func() {
		flushSighting()
		if cur != nil && cur.ObservationID != "" {
			out = append(out, *cur)
		}
		cur = nil
	}

	for _, raw := range strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n") {
		line := strings.TrimSpace(raw)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		key, val := splitKV(line)
		switch key {
		case "OBS":
			flushObservation()
			cur = &ParsedObservation{ObservationID: val, Date: "CURRENT"}
		case "END":
			flushObservation()
		}
		if cur == nil {
			continue
		}
		switch key {
		case "LOC":
			cur.LocationCode = strings.ToUpper(val)
		case "DATE":
			cur.Date = val
		case "SPEC":
			flushSighting()
			curSpec = &ParsedSighting{SpeciesCode: val}
		}
		if curSpec == nil {
			continue
		}
		switch key {
		case "LAT":
			curSpec.Latitude = parseFloatOr(val, 0)
		case "LON":
			curSpec.Longitude = parseFloatOr(val, 0)
		case "ROUTE":
			fields := strings.Fields(val)
			if len(fields) == 0 {
				continue
			}
			r := Route{Start: fields[0]}
			if len(fields) > 1 && fields[1] != "CONTINUOUS" {
				r.End = fields[1]
			}
			curSpec.Route = append(curSpec.Route, r)
		}
	}
	flushObservation()
	return out
}

// FromFile reloads the whole species-sighting inventory from one observation file
// inside a transaction. A sighting with N routes becomes N sighting rows,
// sharing species_code but not sighting_id, one row per migration route.
func FromFile(ctx context.Context, db *sql.DB, path string) (Counts, error) {
	var c Counts
	data, err := os.ReadFile(path)
	if err != nil {
		return c, err
	}
	observations := Parse(data)
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return c, err
	}
	defer tx.Rollback()
	for _, t := range []string{"sightings", "species"} {
		if _, err := tx.ExecContext(ctx, "DELETE FROM "+t); err != nil {
			return c, err
		}
	}
	for _, obs := range observations {
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO species (species_id, location_code, observation_date,
			        latitude, longitude)
			 VALUES (?, ?, ?, ?, ?)`,
			obs.SpeciesCode, nullStr(obs.LocationCode), obs.Date, obs.Latitude, obs.Longitude); err != nil {
			return c, err
		}
		c.Species++
		for _, sp := range obs.Species {
			if len(sp.Route) == 0 {
				continue
			}
			for i, rt := range sp.Route {
				sightingID := obs.SpeciesCode + "-" + strconv.Itoa(i+1)
				if _, err := tx.ExecContext(ctx,
					`INSERT INTO sightings (sighting_id, species_id, route_start, route_end)
					 VALUES (?, ?, ?, ?)`,
					sightingID, obs.SpeciesCode, rt.Start, nullStr(rt.End)); err != nil {
					return c, err
				}
				c.Sightings++
			}
		}
	}
	return c, tx.Commit()
}

// EnsureSeeded loads the seed only when the inventory is empty.
func EnsureSeeded(ctx context.Context, db *sql.DB, path string) (bool, Counts, error) {
	var n int
	if err := db.QueryRowContext(ctx, `SELECT count(*) FROM species`).Scan(&n); err != nil {
		return false, Counts{}, err
	}
	if n != 0 {
		return false, Counts{}, nil
	}
	c, err := FromFile(ctx, db, path)
	return err == nil, c, err
}
