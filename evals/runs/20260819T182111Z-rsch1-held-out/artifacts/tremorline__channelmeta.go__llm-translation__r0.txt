// Package ingest reads the WLS Species Observation Format (.wso) — a
// local profile invented for this system, not SEED, not StationXML, not any
// published observation format. A file is a sequence of sighting blocks. Each
// block opens with an OBS line and a run of KEY value lines describing the
// sighting, then zero or more SPEC blocks describing one species each. A SPEC
// block carries its species details and one or more ROUTE lines, each
// naming a migration route; a species observed during different migrations gets a
// second ROUTE line rather than a second SPEC line. A block closes on an
// explicit END line, which is what separates records — blank lines and '#'
// comment lines are ignored wherever they appear and carry no structural
// meaning.
package ingest

import (
	"context"
	"database/sql"
	"os"
	"strconv"
	"strings"
)

type Counts struct {
	Sightings int `json:"sightings"`
	Species   int `json:"species"`
}

// route is one migration route for a species. End is empty for "CONTINUOUS", the
// route still active.
type Route struct {
	Start, End string
}

type ParsedSpecies struct {
	Name         string
	Description  string
	MigrationRate float64
	Routes       []Route
}

type ParsedSighting struct {
	SightingID, LocationCode, DateTime string
	Species                            []ParsedSpecies
}

func parseFloatOr(s string, fallback float64) float64 {
	v, err := strconv.ParseFloat(strings.TrimSpace(s), 64)
	if err != nil {
		return fallback
	}
	return v
}

func nullStr(s string) any {
	if s == "" {
		return nil
	}
	return s
}

// splitKV separates a line into its leading key and the rest of the line,
// which may itself contain internal whitespace (location and description are
// free text).
func splitKV(line string) (string, string) {
	parts := strings.SplitN(strings.TrimSpace(line), " ", 2)
	key := parts[0]
	val := ""
	if len(parts) == 2 {
		val = strings.TrimSpace(parts[1])
	}
	return key, val
}

// Parse walks the file a line at a time. The sighting and species under
// construction are held in this scope and mutated from inside the key
// switch; flushSpecies and flushSighting take no arguments and read those
// variables directly, so a case arm closes a record blind.
func Parse(data []byte) []ParsedSighting {
	var out []ParsedSighting
	var cur *ParsedSighting
	var curSpec *ParsedSpecies

	flushSpecies := func() {
		if cur != nil && curSpec != nil && curSpec.Name != "" {
			cur.Species = append(cur.Species, *curSpec)
		}
		curSpec = nil
	}
	flushSighting := func() {
		flushSpecies()
		if cur != nil && cur.SightingID != "" {
			out = append(out, *cur)
		}
		cur = nil
	}

	for _, raw := range strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n") {
		line := strings.TrimSpace(raw)
		if line == "" || strings.HasPrefix(line, "#") {
			continue
		}
		key, val := splitKV(line)
		switch key {
		case "OBS":
			flushSighting()
			cur = &ParsedSighting{SightingID: val, DateTime: "CURRENT"}
		case "END":
			flushSighting()
		}
		if cur == nil {
			continue
		}
		switch key {
		case "LOC":
			cur.LocationCode = strings.ToUpper(val)
		case "DESC":
			cur.Description = val
		case "DATE":
			cur.DateTime = val
		case "SPEC":
			flushSpecies()
			curSpec = &ParsedSpecies{Name: val}
		}
		if curSpec == nil {
			continue
		}
		switch key {
		case "RATE":
			curSpec.MigrationRate = parseFloatOr(val, 0)
		case "ROUTE":
			fields := strings.Fields(val)
			if len(fields) == 0 {
				continue
			}
			r := Route{Start: fields[0]}
			if len(fields) > 1 && fields[1] != "CONTINUOUS" {
				r.End = fields[1]
			}
			curSpec.Routes = append(curSpec.Routes, r)
		}
	}
	flushSighting()
	return out
}

// FromFile reloads the whole species-sighting inventory from one observation file
// inside a transaction. A species with N routes becomes N species rows,
// sharing species_name but not species_id, one row per migration window.
func FromFile(ctx context.Context, db *sql.DB, path string) (Counts, error) {
	var c Counts
	data, err := os.ReadFile(path)
	if err != nil {
		return c, err
	}
	sightings := Parse(data)
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return c, err
	}
	defer tx.Rollback()
	for _, t := range []string{"species", "sightings"} {
		if _, err := tx.ExecContext(ctx, "DELETE FROM "+t); err != nil {
			return c, err
		}
	}
	for _, st := range sightings {
		if _, err := tx.ExecContext(ctx,
			`INSERT INTO sightings (sighting_id, location_code, date_time)
			 VALUES (?, ?, ?)`,
			st.SightingID, nullStr(st.LocationCode), st.DateTime); err != nil {
			return c, err
		}
		c.Sightings++
		for _, sp := range st.Species {
			if len(sp.Routes) == 0 {
				continue
			}
			for i, rp := range sp.Routes {
				speciesID := st.SightingID + "-" + sp.Name + "-" + strconv.Itoa(i+1)
				if _, err := tx.ExecContext(ctx,
					`INSERT INTO species (species_id, sighting_id, species_name,
					        migration_rate, route_start, route_end)
					 VALUES (?, ?, ?, ?, ?, ?)`,
					speciesID, st.SightingID, sp.Name, sp.MigrationRate, rp.Start, nullStr(rp.End)); err != nil {
					return c, err
				}
				c.Species++
			}
		}
	}
	return c, tx.Commit()
}

// EnsureSeeded loads the seed only when the inventory is empty.
func EnsureSeeded(ctx context.Context, db *sql.DB, path string) (bool, Counts, error) {
	var n int
	if err := db.QueryRowContext(ctx, `SELECT count(*) FROM sightings`).Scan(&n); err != nil {
		return false, Counts{}, err
	}
	if n != 0 {
		return false, Counts{}, nil
	}
	c, err := FromFile(ctx, db, path)
	return err == nil, c, err
}
