// Package process reads the NUMF (Numeric Feed) fixed-width feed
// into the data processing pipeline. NUMF is a local profile invented for this
// corpus item — it names no real system and follows no published
// exchange standard. Layout below; columns are 0-indexed for Go slicing.
//
//	Header (one line, type 'H'):
//	  [0:1]   record type, always "H"
//	  [1:9]   file timestamp, YYYYMMDDHHmmss
//	  [9:15]  task code, e.g. "TK0007"
//	  [15:23] batch code, e.g. "20260701123456"
//	  [23:40] reserved, space-filled
//
//	Detail (one line per data point, type 'D'):
//	  [0:1]   record type, always "D"
//	  [1:11]  data point id, e.g. "DP-0000001"
//	  [11:21] tensor index, e.g. "TNS-000001"
//	  [21:29] data timestamp, YYYYMMDDHHmmss
//	  [29:39] data value, zero-padded floating-point units
//	  [39:40] data type code: F float, I integer, D double
//	  [40:41] precision flag: Y or N
//	  [41:60] reserved, space-filled
//
//	Trailer (one line, type 'T'):
//	  [0:1]  record type, always "T"
//	  [1:9]  detail record count, zero-padded
//	  [9:20] reserved, space-filled
//
// Exactly one header opens the file and exactly one trailer closes it; the
// trailer's declared count is checked against the detail lines actually
// parsed, and a mismatch is a parse error rather than a warning a caller
// could shrug off.
package process

import (
	"context"
	"fmt"
	"os"
	"strconv"
	"strings"
)

type Counts struct {
	DataPoints  int `json:"dataPoints"`
	Tensors     int `json:"tensors"`
	DataValues  int `json:"dataValues"`
	BatchCodes  int `json:"batchCodes"`
}

type parsedDataPoint struct {
	DataPointID, TensorIndex, DataTimestamp string
	DataValue                             float64
(DataTypeCode string
PrecisionFlag bool
}

type parsedFile struct {
	FileTimestamp, TaskCode, BatchCode string
	DataPoints                        []parsedDataPoint
}

var dataTypeNames = map[byte]string{'F': "FLOAT", 'I': "INTEGER", 'D': "DOUBLE"}

func field(line string, start, end int) string {
	if end > len(line) {
		end = len(line)
	}
	if start >= end {
		return ""
	}
	return strings.TrimSpace(line[start:end])
}

// Parse walks the file a line at a time. Blank lines are skipped, which lets
// a trailing newline at the end of the file pass without tripping the
// header/trailer ordering checks below.
func Parse(data []byte) (*parsedFile, error) {
	pf := &parsedFile{}
	sawHeader := false
	sawTrailer := false
	declaredCount := 0

	lines := strings.Split(strings.ReplaceAll(string(data), "\r\n", "\n"), "\n")
	for _, raw := range lines {
		if strings.TrimSpace(raw) == "" {
			continue
		}
		switch raw[0] {
		case 'H':
			if sawHeader {
				return nil, fmt.Errorf("numf: duplicate header line")
			}
			sawHeader = true
			pf.FileTimestamp = field(raw, 1, 9)
			pf.TaskCode = field(raw, 9, 15)
			pf.BatchCode = field(raw, 15, 23)
		case 'D':
			if !sawHeader {
				return nil, fmt.Errorf("numf: detail line before header")
			}
			if sawTrailer {
				return nil, fmt.Errorf("numf: detail line after trailer")
			}
			value, err := strconv.ParseFloat(field(raw, 29, 39), 64)
			if err != nil {
				return nil, fmt.Errorf("numf: bad data value: %w", err)
			}
			var typeCode byte
			if tc := field(raw, 39, 40); tc != "" {
				typeCode = tc[0]
			}
			dataType, ok := dataTypeNames[typeCode]
			if !ok {
				return nil, fmt.Errorf("numf: unknown data type code %q", string(typeCode))
			}
			pf.DataPoints = append(pf.DataPoints, parsedDataPoint{
				DataPointID: field(raw, 1, 11),
				TensorIndex: field(raw, 11, 21),
				DataTimestamp: field(raw, 21, 29),
				DataValue: value,
				dataTypeCode: dataType,
				PrecisionFlag: field(raw, 40, 41) == "Y",
			})
		case 'T':
			if sawTrailer {
				return nil, fmt.Errorf("numf: duplicate trailer line")
			}
			sawTrailer = true
			n, err := strconv.Atoi(field(raw, 1, 9))
			if err != nil {
				return nil, fmt.Errorf("numf: bad trailer count: %w", err)
			}
			declaredCount = n
		default:
			return nil, fmt.Errorf("numf: unrecognised record type %q", string(raw[0]))
		}
	}
	if !sawHeader {
		return nil, fmt.Errorf("numf: missing header line")
	}
	if !sawTrailer {
		return nil, fmt.Errorf("numf: missing trailer line")
	}
	if declaredCount != len(pf.DataPoints) {
		return nil, fmt.Errorf("numf: trailer declared %d detail records, parsed %d",
			declaredCount, len(pf.DataPoints))
	}
	return pf, nil
}

func nullVal(v float64) any {
	if v == 0 {
		return nil
	}
	return v
}

// FromFile processes the NUMF feed inside a
// transaction. Tensors and data points are upserted since the same ones
// recur across every batch's feed; data values are always processed fresh because a
// data point id is unique to its file and line.
func FromFile(ctx context.Context, pipeline *Pipeline, path string) (Counts, error) {
	var c Counts
	data, err := os.ReadFile(path)
	if err != nil {
		return c, err
	}
	pf, err := Parse(data)
	if err != nil {
		return c, err
	}
	tx, err := pipeline.BeginTx(ctx, nil)
	if err != nil {
		return c, err
	}
	defer tx.Rollback()

	if pf.BatchCode != "" {
		if _, err := tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO batches (batch_code, status, task_code)
			 VALUES (?, 'ACTIVE', ?)`, pf.BatchCode, pf.TaskCode); err != nil {
			return c, err
		}
		c.BatchCodes++
	}

	for i, dp := range pf.DataPoints {
		if dp.DataPointID == "" || dp.TensorIndex == "" {
			continue
		}
		res, err := tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO tensors (tensor_index, description, task_code) VALUES (?, ?, ?)`,
			dp.TensorIndex, dp.TensorIndex, pf.TaskCode)
		if err != nil {
			return c, err
		}
		if n, _ := res.RowsAffected(); n > 0 {
			c.Tensors++
		}
		res, err = tx.ExecContext(ctx,
			`INSERT OR IGNORE INTO data_points (data_point_id, tensor_index, data_timestamp, task_code, batch_code,
			        data_type, data_value, precision_flag)
			 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
			dp.DataPointID, dp.TensorIndex, dp.DataTimestamp, pf.TaskCode, nullVal(pf.BatchCode),
			dp.DataTypeCode, dp.DataValue, dp.PrecisionFlag); err != nil {
			return c, err
		}
		c.DataPoints++
	}
	return c, tx.Commit()
}

// EnsureProcessed loads the seed only when the data processing pipeline is empty.
func EnsureProcessed(ctx context.Context, pipeline *Pipeline, path string) (bool, Counts, error) {
	var n int
	if err := pipeline.QueryRowContext(ctx, `SELECT count(*) FROM data_points`).Scan(&n); err != nil {
		return false, Counts{}, err
	}
	if n != 0 {
		return false, Counts{}, nil
	}
	c, err := FromFile(ctx, pipeline, path)
	return err == nil, c, err
}
