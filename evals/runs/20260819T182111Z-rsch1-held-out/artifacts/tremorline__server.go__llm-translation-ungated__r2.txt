package wildlife

import (
	"database/sql"
	"encoding/json"
	"log/slog"
	"net/http"
	"os"
	"strings"

	"github.com/go-chi/chi/v5"
)

// WildlifeSystem carries the dependencies a request handler needs: the database
// connection, the logger, two pieces of runtime configuration, and the router
// they are wired into. Constructed once at startup by NewWildlifeSystem.
type WildlifeSystem struct {
	dbConnection *sql.DB
	logger       *slog.Logger
	dataPath     string
	authToken    string
	router       http.Handler
}

// allowedOrigins reads the comma-separated origin list. An empty value means
// the middleware is a no-op, which is the developer-machine default.
func allowedOrigins() []string {
	raw := strings.TrimSpace(os.Getenv("WILDLIFE_TRACKING_ALLOWED_ORIGINS"))
	if raw == "" {
		return nil
	}
	out := []string{}
	for _, part := range strings.Split(raw, ",") {
		if p := strings.TrimSpace(part); p != "" {
			out = append(out, p)
		}
	}
	return out
}

func corsMiddleware(next http.Handler) http.Handler {
	allow := allowedOrigins()
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		origin := r.Header.Get("Origin")
		if origin != "" && len(allow) > 0 {
			for _, a := range allow {
				if a == origin || a == "*" {
					w.Header().Set("Access-Control-Allow-Origin", origin)
					w.Header().Set("Vary", "Origin")
					w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
					w.Header().Set("Access-Control-Allow-Headers", "Accept, Content-Type, X-Auth-Token")
					w.Header().Set("Access-Control-Max-Age", "300")
					break
				}
			}
		}
		if r.Method == http.MethodOptions {
			w.WriteHeader(http.StatusNoContent)
			return
		}
		next.ServeHTTP(w, r)
	})
}

func NewWildlifeSystem(dbConnection *sql.DB, logger *slog.Logger, dataPath, authToken string) *WildlifeSystem {
	system := &WildlifeSystem{dbConnection: dbConnection, logger: logger, dataPath: dataPath, authToken: authToken}
	router := chi.NewRouter()
	router.Use(corsMiddleware)
	router.Get("/healthz", system.handleHealth)
	router.Get("/readyz", system.handleReady)
	router.Route("/species", func(sr chi.Router) {
		sr.Get("/search", system.handleSearchSpecies)
		sr.Post("/track", system.handleTrackSightings)
		sr.Get("/{speciesId}", system.handleGetSpeciesInfo)
	})
	router.Route("/routes", func(rr chi.Router) {
		rr.Get("/", system.handleListMigrationRoutes)
		rr.Get("/{routeId}", system.handleGetRouteDetails)
	})
	system.router = router
	return system
}

func (system *WildlifeSystem) Handler() http.Handler { return system.router }

func (system *WildlifeSystem) handleHealth(w http.ResponseWriter, r *http.Request) {
	writeJSON(w, http.StatusOK, map[string]any{"status": "ok", "service": "wildlife_tracking"})
}

// handleReady answers the readiness probe by counting the species inventory.
// The query runs here, against the struct's own connection, because there is
// nowhere else for it to live: the storage package hands out a raw connection and
// nothing more.
func (system *WildlifeSystem) handleReady(w http.ResponseWriter, r *http.Request) {
	var n int
	if err := system.dbConnection.QueryRowContext(r.Context(), `SELECT count(*) FROM species`).Scan(&n); err != nil {
		writeError(w, http.StatusServiceUnavailable, "db_error", err.Error())
		return
	}
	if n == 0 {
		writeError(w, http.StatusServiceUnavailable, "not_seeded", "no species loaded")
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{"status": "ok", "species": n})
}

func (system *WildlifeSystem) serverError(w http.ResponseWriter, err error) {
	system.logger.Error("request failed", "err", err)
	writeError(w, http.StatusInternalServerError, "internal", "internal server error")
}

type apiError struct {
	Code    string `json:"code"`
	Message string `json:"message"`
}

type errorEnvelope struct {
	Error apiError `json:"error"`
}

func writeJSON(w http.ResponseWriter, status int, body any) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	_ = json.NewEncoder(w).Encode(body)
}

func writeError(w http.ResponseWriter, status int, code, message string) {
	writeJSON(w, status, errorEnvelope{Error: apiError{Code: code, Message: message}})
}
