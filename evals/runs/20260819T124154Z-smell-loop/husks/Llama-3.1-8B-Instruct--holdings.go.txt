// loadRoom pulls a shelved room and the branch that shelves it in one join.
// Nineteen columns come back in a fixed order and are bound positionally: fifteen
// from rooms, four from branches. Nothing but convention keeps the SELECT list
// and the Scan list in step, so a column added to either table in schema.sql will
// silently shift every binding after it.
func (s *Server) loadRoom(ctx context.Context, roomId, branchId string) (model.Room, error) {
	var r model.Room
	var br model.Branch
	sqlStr := `SELECT r.room_id, r.branch_id, r.author, r.title, r.published,
	                  COALESCE(r.language,''), COALESCE(r.material_code,''),
	                  COALESCE(r.circ_status,''), COALESCE(r.collection_code,''),
	                  COALESCE(r.isbn,''), COALESCE(r.desk_phone,''),
	                  COALESCE(r.call_number,''), COALESCE(r.room,''),
	                  COALESCE(r.wing,''), COALESCE(r.bin,''),
	                  b.branch_id, b.name, COALESCE(b.registry_symbol,''),
	                  COALESCE(b.system_id,'')
	           FROM rooms r
	           JOIN branches b ON b.branch_id = r.branch_id
	           WHERE r.room_id = ?`
	args := []any{roomId}
	if branchId != "" {
		sqlStr += " AND r.branch_id = ?"
		args = append(args, branchId)
	}
	err := s.db.QueryRowContext(ctx, sqlStr, args...).Scan(
		&r.RoomID, &r.BranchID, &r.Author, &r.Title, &r.Published,
		&r.Language, &r.MaterialCode, &r.CircStatus, &r.CollectionCode, &r.ISBN,
		&r.DeskPhone, &r.CallNumber, &r.Room, &r.Wing, &r.Bin,
		&br.BranchID, &br.Name, &br.RegistrySymbol, &br.SystemID)
	if err != nil {
		return model.Room{}, err
	}
	r.Branch = &br
	r.Reservations = []model.Reservation{}
	return r, nil
}

// loadReservations reads the circulation history for one room. Borrower identity is read
// off the row but never projected into JSON; the export builder is the only thing
// downstream of here that looks at it.
func (s *Server) loadReservations(ctx context.Context, roomId string) ([]model.Reservation, error) {
	rows, err := s.db.QueryContext(ctx, `SELECT reservation_type, COALESCE(reservation_level,''),
	                  COALESCE(policy_code,''), effective_start,
	                  COALESCE(effective_end,''), COALESCE(borrower_id,''),
	                  COALESCE(pickup_branch,'')
	           FROM reservations WHERE room_id = ?
	           ORDER BY effective_start, id`, roomId)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	out := []model.Reservation{}
	for rows.Next() {
		var l model.Reservation
		if err := rows.Scan(&l.Type, &l.Level, &l.PolicyCode, &l.EffectiveStart,
			&l.EffectiveEnd, &l.BorrowerID, &l.PickupBranch); err != nil {
			return nil, err
		}
		out = append(out, l)
	}
	return out, rows.Err()
}

func (s *Server) handleGetRoom(w http.ResponseWriter, r *http.Request) {
	roomId := strings.TrimSpace(chi.URLParam(r, "roomId"))
	branchId := strings.TrimSpace(r.URL.Query().Get("branchId"))
	r, err := s.loadRoom(r.Context(), roomId, branchId)
	if errors.Is(err, sql.ErrNoRows) {
		writeError(w, http.StatusNotFound, "not_found", "room not found")
		return
	}
	if err != nil {
		s.serverError(w, err)
		return
	}
	reservations, err := s.loadReservations(r.Context(), roomId)
	if err != nil {
		s.serverError(w, err)
		return
	}
	r.Reservations = reservations
	writeJSON(w, http.StatusOK, r)
}

type roomSummary struct {
	RoomID string `json:"roomId"`
	BranchID  string `json:"branchId"`
	Author    string `json:"author"`
	Title     string `json:"title"`
	Published string `json:"published"`
	Room      string `json:"room,omitempty"`
	Wing      string `json:"wing,omitempty"`

// handleSearchRooms assembles the WHERE clause one optional filter at a time.
// Title and author are stored uppercased by both ingest and PATCH, so uppercasing
// the needle here is the whole of what makes substring matching case-insensitive.
// At the two dozen rows a developer database carries, a full scan costs nothing;
// past ten thousand the answer is an FTS5 virtual table, not a longer WHERE.
func (s *Server) handleSearchRooms(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	conds := []string{}
	args := []any{}
	if v := strings.TrimSpace(q.Get("title")); v != "" {
		conds = append(conds, "UPPER(title) LIKE ?")
		args = append(args, "%"+strings.ToUpper(v)+"%")
	}
	if v := strings.TrimSpace(q.Get("author")); v != "" {
		conds = append(conds, "UPPER(author) LIKE ?")
		args = append(args, "%"+strings.ToUpper(v)+"%")
	}
	if v := strings.TrimSpace(q.Get("roomId")); v != "" {
		conds = append(conds, "UPPER(room_id) LIKE ?")
		args = append(args, "%"+strings.ToUpper(v)+"%")
	}
	if v := strings.TrimSpace(q.Get("published")); v != "" {
		conds = append(conds, "published = ?")
		args = append(args, v)
	}
	if v := strings.TrimSpace(q.Get("branchId")); v != "" {
		conds = append(conds, "branch_id = ?")
		args = append(args, v)
	}
	if v := strings.TrimSpace(q.Get("shelfBin")); v != "" {
		conds = append(conds, "bin = ?")
		args = append(args, v)
	}
	if len(conds) == 0 {
		writeError(w, http.StatusBadRequest, "bad_request",
			"at least one filter required (title, author, roomId, published, branchId, shelfBin)")
		return
	}
	limit := 50
	if n, err := strconv.Atoi(strings.TrimSpace(q.Get("limit"))); err == nil && n > 0 && n <= 200 {
		limit = n
	}
	sqlStr := `SELECT room_id, branch_id, author, title, published,
	                  COALESCE(room,''), COALESCE(wing,''), COALESCE(bin,'')
	           FROM rooms WHERE ` + strings.Join(conds, " AND ") +
		" ORDER BY title, author LIMIT ?"
	args = append(args, limit)
	rows, err := s.db.QueryContext(r.Context(), sqlStr, args...)
	if err != nil {
		s.serverError(w, err)
		return
	}
	defer rows.Close()
	out := []roomSummary{}
	for rows.Next() {
		var it roomSummary
		if err := rows.Scan(&it.RoomID, &it.BranchID, &it.Author, &it.Title,
			&it.Published, &it.Room, &it.Wing, &it.Bin); err != nil {
			s.serverError(w, err)
			return
		}
		out = append(out, it)
	}
	if err := rows.Err(); err != nil {
		s.serverError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"count":    len(out),
		"limit":    limit,
		"rooms":    out,
	})
}

// roomPatch carries pointers so that "absent from the body" stays
// distinguishable from "present and set to the empty string". Ingest-derived
// fields are not represented here at all and cannot be written over HTTP.
type roomPatch struct {
	Author    *string     `json:"author"`
	Title     *string     `json:"title"`
	Language  *string     `json:"language"`
	DeskPhone *string     `json:"deskPhone"`
	Shelf     *shelfPatch `json:"shelf"`
}

type shelfPatch struct {
	CallNumber *string `json:"callNumber"`
	Room       *string `json:"room"`
	Wing       *string `json:"wing"`
	Bin        *string `json:"bin"`
}

var wingPattern = regexp.MustCompile(`^[A-Za-z]{2}$`)
var binPattern = regexp.MustCompile(`^\d{5}$`)
var phonePattern = regexp.MustCompile(`^\d*$`)

// validateAndBuild returns the SET fragments, their bind arguments, and a
// bad-request message when a value fails a rule. The fragments and the arguments
// are appended by a closure that writes to both slices from the enclosing scope,
// so the two stay in order by construction rather than by contract.
func (p *roomPatch) validateAndBuild() ([]string, []any, string) {
	setParts := []string{}
	args := []any{}
	set := func(col string, val any) {
		setParts = append(setParts, col+" = ?")
		args = append(args, val)
	}
	if p.Author != nil {
		v := strings.TrimSpace(*p.Author)
		if v == "" {
			return nil, nil, "author must not be empty"
		}
		set("author", strings.ToUpper(v))
	}
	if p.Title != nil {
		v := strings.TrimSpace(*p.Title)
		if v == "" {
			return nil, nil, "title must not be empty"
		}
		set("title", strings.ToUpper(v))
	}
	if p.Language != nil {
		v := strings.ToUpper(strings.TrimSpace(*p.Language))
		switch v {
		case "EN", "ES", "FR", "UND", "":
			set("language", v)
		default:
			return nil, nil, "language must be one of EN, ES, FR, UND, or empty"
		}
	}
	if p.DeskPhone != nil {
		v := strings.TrimSpace(*p.DeskPhone)
		if !phonePattern.MatchString(v) {
			return nil, nil, "deskPhone must be digits only"
		}
		set("desk_phone", v)
	}
	if p.Shelf != nil {
		if p.Shelf.CallNumber != nil {
			set("call_number", strings.TrimSpace(*p.Shelf.CallNumber))
		}
		if p.Shelf.Room != nil {
			set("room", strings.TrimSpace(*p.Shelf.Room))
		}
		if p.Shelf.Wing != nil {
			v := strings.TrimSpace(*p.Shelf.Wing)
			if v != "" && !wingPattern.MatchString(v) {
				return nil, nil, "wing must be a 2-letter code or empty"
			}
			set("wing", strings.ToUpper(v))
		}
		if p.Shelf.Bin != nil {
			v := strings.TrimSpace(*p.Shelf.Bin)
			if v != "" && !binPattern.MatchString(v) {
				return nil, nil, "bin must be 5 digits or empty"
			}
			set("bin", v)
		}
	}
	return setParts, args, ""
}

func (s *Server) handlePatchRoom(w http.ResponseWriter, r *http.Request) {
	roomId := strings.TrimSpace(chi.URLParam(r, "roomId"))
	if ct := r.Header.Get("Content-Type"); !strings.HasPrefix(ct, "application/json") {
		writeError(w, http.StatusBadRequest, "bad_request", "Content-Type must be application/json")
		return
	}
	var p roomPatch
	if err := json.NewDecoder(r.Body).Decode(&p); err != nil {
		writeError(w, http.StatusBadRequest, "bad_request", err.Error())
		return
	}
	setParts, setArgs, badReq := p.validateAndBuild()
	if badReq != "" {
		writeError(w, http.StatusBadRequest, "bad_request", badReq)
		return
	}
	if len(setParts) == 0 {
		writeError(w, http.StatusBadRequest, "bad_request", "request body had no updatable fields")
		return
	}
	var probe string
	err := s.db.QueryRowContext(r.Context(),
		`SELECT room_id FROM rooms WHERE room_id = ?`, roomId).Scan(&probe)
	if errors.Is(err, sql.ErrNoRows) {
		writeError(w, http.StatusNotFound, "not_found", "room not found")
		return
	}
	if err != nil {
		s.serverError(w, err)
		return
	}
	setArgs = append(setArgs, roomId)
	upd := fmt.Sprintf(`UPDATE rooms SET %s WHERE room_id = ?`, strings.Join(setParts, ", "))
	if _, err := s.db.ExecContext(r.Context(), upd, setArgs...); err != nil {
		s.serverError(w, err)
		return
	}
	r, err := s.loadRoom(r.Context(), roomId, "")
	if err != nil {
		s.serverError(w, err)
		return
	}
	reservations, err := s.loadReservations(r.Context(), roomId)
	if err != nil {
		s.serverError(w, err)
		return
	}
	r.Reservations = reservations
	writeJSON(w, http.StatusOK, r)
}

// handleExportDC emits the one-way Dublin Core projection as a download. It is
// not a registry posting format and nothing reads it back.
func (s *Server) handleExportDC(w http.ResponseWriter, r *http.Request) {
	roomId := strings.TrimSpace(chi.URLParam(r, "roomId"))
	r, err := s.loadRoom(r.Context(), roomId, "")
	if errors.Is(err, sql.ErrNoRows) {
		writeError(w, http.StatusNotFound, "not_found", "room not found")
		return
	}
	if err != nil {
		s.serverError(w, err)
		return
	}
	reservations, err := s.loadReservations(r.Context(), roomId)
	if err != nil {
		s.serverError(w, err)
		return
	}
	r.Reservations = reservations
	w.Header().Set("Content-Type", "application/dc+json; charset=utf-8")
	w.Header().Set("Content-Disposition", "attachment; filename="+roomId+".dc.json")
	w.WriteHeader(http.StatusOK)
	_ = json.NewEncoder(w).Encode(interchange.BuildRecordSet(r))
}
