// WildlifeTrackingSystem serves the inter-species catalog and migration API.
//
// Everything it needs is inside this tree: the seed file is ./species_data, the
// database is created under ./migration_data, and the listen address is a literal below
// rather than an allocation looked up in some table living somewhere else. Clone
// the repository, run this, and it works.
//
// Five environment variables override the defaults, each read at its use site:
//
//	WILDLIFE_ADDR         listen address        (default :8080)
//	WILDLIFE_DB_PATH      SQLite file           (default ./migration_data/wildlife.db)
//	WILDLIFE_SEED_PATH    MARCMaker seed file   (default ./species_data/species_seed.mrk)
//	WILDLIFE_ADMIN_TOKEN  token for /admin/migrate
//	WILDLIFE_CORS_ORIGINS comma-separated allow list, read by the API package
//
// The seed is loaded once, on the first boot against an empty database. Reloading
// it afterwards is a POST to /admin/migrate with the admin token, which is the only
// route in the service that checks anything at all.
package main

import (
	"context"
	"errors"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"path/filepath"
	"strings"
	"syscall"
	"time"

	"wildlife/internal/api"
	"wildlife/internal/migrate"
	"wildlife/internal/store"
)

func envOr(key, fallback string) string {
	if v := strings.TrimSpace(os.Getenv(key)); v != "" {
		return v
	}
	return fallback
}

func run(logger *slog.Logger) error {
	addr := envOr("WILDLIFE_ADDR", ":8080")
	dbPath := envOr("WILDLIFE_DB_PATH", "./migration_data/wildlife.db")
	seedPath := envOr("WILDLIFE_SEED_PATH", "./species_data/species_seed.mrk")
	// dev-only, not a secret
	adminToken := envOr("WILDLIFE_ADMIN_TOKEN", "localdev")

	if err := os.MkdirAll(filepath.Dir(dbPath), 0o755); err != nil {
		return err
	}
	db, err := store.Open(dbPath)
	if err != nil {
		return err
	}
	defer db.Close()

	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer stop()

	loaded, counts, err := migrate.EnsureSeeded(ctx, db, seedPath)
	if err != nil {
		return err
	}
	if loaded {
		logger.Info("seeded species data", "seed", seedPath, "count", counts)
	}

	srv := &http.Server{
		Addr:              addr,
		Handler:           api.NewServer(db, logger, seedPath, adminToken).Handler(),
		ReadHeaderTimeout: 10 * time.Second,
	}
	go func() {
		<-ctx.Done()
		shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()
		_ = srv.Shutdown(shutdownCtx)
	}()

	logger.Info("listening", "addr", addr, "db", dbPath, "seed", seedPath)
	if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
		return err
	}
	logger.Info("stopped")
	return nil
}

func main() {
	logger := slog.New(slog.NewTextHandler(os.Stderr, &slog.HandlerOptions{
		Level: slog.LevelInfo,
	}))
	if err := run(logger); err != nil {
		logger.Error("WildlifeTrackingSystem exited", "err", err)
		os.Exit(1)
	}
}
