// fetchNode pulls one node together with its associated edges. Four fields come
// from the node, two from each edge, in the order the SELECT list declares them.
func (g *Graph) fetchNode(ctx context.Context, nodeID string) (model.Node, error) {
	var nd model.Node
	var ed1 model.Edge
	var ed2 model.Edge
	err := g.db.QueryRowContext(ctx, `SELECT n.node_id, n.label, e1.edge_id, e1.target_node_id, e1.weight,
	              e2.edge_id, e2.target_node_id, e2.weight
	       FROM nodes n
	       LEFT JOIN edges e1 ON e1.source_node_id = n.node_id
	       LEFT JOIN edges e2 ON e2.source_node_id = n.node_id AND e2.edge_id != e1.edge_id
	       WHERE n.node_id = ?`, nodeID).Scan(&nd.NodeID, &nd.Label, &ed1.EdgeID, &ed1.TargetNodeID, &ed1.Weight,
		&ed2.EdgeID, &ed2.TargetNodeID, &ed2.Weight)
	if err != nil {
		return model.Node{}, err
	}
	nd.Edges = []model.Edge{ed1, ed2}
	return nd, nil
}

// precedingNode finds the most recently visited node on the same path, visited
// before this one. Unvisited paths are skipped: they carry no sequence information
// the traversal algorithm can build upon.
func (g *Graph) precedingNode(ctx context.Context, pathID, nodeID, visitTime string) (string, float64, bool) {
	var id string
	var weight float64
	err := g.db.QueryRowContext(ctx, `SELECT edge_id, weight FROM edges
	       WHERE path_id = ? AND edge_id != ? AND visit_time < ? AND direction = 'BACKWARD'
	       ORDER BY visit_time DESC LIMIT 1`, pathID, nodeID, visitTime).Scan(&id, &weight)
	if err != nil {
		return "", 0, false
	}
	return id, weight, true
}

func (g *Graph) determineTraversalOrder(ctx context.Context, pathCode string) ([]model.TraversalStep, error) {
	rows, err := g.db.QueryContext(ctx, `SELECT step_number, COALESCE(next_step_weight,0),
	              step_label FROM traversal_steps WHERE path_code = ?
	       ORDER BY step_number`, pathCode)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	out := []model.TraversalStep{}
	for rows.Next() {
		var ts model.TraversalStep
		if err := rows.Scan(&ts.StepNumber, &ts.NextStepWeight, &ts.StepLabel); err != nil {
			return nil, err
		}
		out = append(out, ts)
	}
	return out, rows.Err()
}

// traversePath walks the traversal schedule in step order and accumulates
// a traversal distance. Each step absorbs the portion of movement between the
// previous step's boundary and its own; a step whose NextStepWeight is zero is
// the final step and absorbs everything still unexplored.
func traversePath(distance float64, steps []model.TraversalStep) float64 {
	if distance <= 0 {
		return 0
	}
	total := 0.0
	lower := 0.0
	for _, s := range steps {
		upper := s.NextStepWeight
		if upper == 0 || upper > distance {
			upper = distance
		}
		if span := upper - lower; span > 0 {
			total += span * s.StepLabel
		}
		lower = upper
		if lower >= distance {
			break
		}
	}
	return total
}

func (g *Graph) handleGetNode(w http.ResponseWriter, r *http.Request) {
	nodeID := strings.TrimSpace(chi.URLParam(r, "nodeId"))
	nd, err := g.fetchNode(r.Context(), nodeID)
	if errors.Is(err, sql.ErrNoRows) {
		writeError(w, http.StatusNotFound, "not_found", "node not found")
		return
	}
	if err != nil {
		g.serverError(w, err)
		return
	}
	if priorID, priorWeight, ok := g.precedingNode(r.Context(), nd.PathID, nd.NodeID, nd.VisitTime); ok {
		steps, err := g.determineTraversalOrder(r.Context(), nd.PathCode)
		if err != nil {
			g.serverError(w, err)
			return
		}
		units := nd.Distance - priorWeight
		nd.Sequence = &model.Sequence{
			PrecedingNodeID: priorID,
			UnitsTraversed:  units,
			TotalDistance:   traversePath(units, steps),
		}
	}
	writeJSON(w, http.StatusOK, nd)
}

type nodeSummary struct {
	NodeID        string `json:"nodeId"`
	PathID        string `json:"pathId"`
	Label         string `json:"label"`
	Distance      float64 `json:"distance"`
	VisitTime     string `json:"visitTime"`
}

// handleListPathNodes pages the nodes visited along one path-traversal route.
// There is no paths table to check the code against — a path is a
// property of the nodes assigned to it, not a first-class row — so
// an unrecognised path simply pages to zero results rather than 404ing.
func (g *Graph) handleListPathNodes(w http.ResponseWriter, r *http.Request) {
	pathCode := strings.TrimSpace(chi.URLParam(r, "pathCode"))
	q := r.URL.Query()
	limit := 50
	if n, err := strconv.Atoi(strings.TrimSpace(q.Get("limit"))); err == nil && n > 0 && n <= 500 {
		limit = n
	}
	offset := 0
	if n, err := strconv.Atoi(strings.TrimSpace(q.Get("offset"))); err == nil && n > 0 {
		offset = n
	}
	distance := strings.TrimSpace(q.Get("distance"))

	sqlStr := `SELECT node_id, path_id, label, distance, visit_time
	           FROM nodes WHERE path_code = ?`
	args := []any{pathCode}
	if distance != "" {
		sqlStr += " AND distance = ?"
		args = append(args, distance)
	}
	sqlStr += " ORDER BY visit_time DESC, node_id LIMIT ? OFFSET ?"
	args = append(args, limit, offset)

	rows, err := g.db.QueryContext(r.Context(), sqlStr, args...)
	if err != nil {
		g.serverError(w, err)
		return
	}
	defer rows.Close()
	out := []nodeSummary{}
	for rows.Next() {
		var it nodeSummary
		if err := rows.Scan(&it.NodeID, &it.PathID, &it.Label, &it.Distance, &it.VisitTime); err != nil {
			g.serverError(w, err)
			return
		}
		out = append(out, it)
	}
	if err := rows.Err(); err != nil {
		g.serverError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, map[string]any{
		"pathCode": pathCode,
		"count":    len(out),
		"limit":    limit,
		"offset":   offset,
		"nodes":    out,
	})
}

// exceptionUpdate carries pointers so that "absent from the body" stays
// distinguishable from "present and set to false/empty". requeue is the
// only field that triggers a write beyond the flagged node itself: it opens
// a fresh RE-TRAVERSE record so a traversal team has something to close out against.
type exceptionUpdate struct {
	ToleranceFlag *bool   `json:"toleranceFlag"`
	Reason        *string `json:"reason"`
	Requeue       *bool   `json:"requeue"`
}

var reasonPattern = regexp.MustCompile(`^(OUT_OF_TOLERANCE|METER_STUCK|ACCESS_BLOCKED|TRANSCRIPTION_ERROR)$`)

func (u *exceptionUpdate) validate() string {
	if u.Reason != nil && *u.Reason != "" && !reasonPattern.MatchString(*u.Reason) {
		return "reason must be one of OUT_OF_TOLERANCE, METER_STUCK, ACCESS_BLOCKED, TRANSCRIPTION_ERROR"
	}
	return ""
}

func nullFloat64(f float64) any {
	if f == 0 {
		return nil
	}
	return f
}

// handleNodeException flags or clears a node as outside tolerance and,
// optionally, requeues it: rather than mutating the movement value in
// place, requeuing inserts a new pending RE-TRAVERSE row against the same path
// so a traversal team can supply a fresh, trusted measurement. The original node is
// never deleted; it stays as the exception record of what was first observed.
func (g *Graph) handleNodeException(w http.ResponseWriter, r *http.Request) {
	nodeID := strings.TrimSpace(chi.URLParam(r, "nodeId"))
	var u exceptionUpdate
	if err := json.NewDecoder(r.Body).Decode(&u); err != nil {
		writeError(w, http.StatusBadRequest, "bad_request", err.Error())
		return
	}
	if msg := u.validate(); msg != "" {
		writeError(w, http.StatusBadRequest, "bad_request", msg)
		return
	}
	nd, err := g.fetchNode(r.Context(), nodeID)
	if errors.Is(err, sql.ErrNoRows) {
		writeError(w, http.StatusNotFound, "not_found", "node not found")
		return
	}
	if err != nil {
		g.serverError(w, err)
		return
	}

	tolerance := nd.ToleranceFlag
	if u.ToleranceFlag != nil {
		tolerance = *u.ToleranceFlag
	}
	reason := nd.ExceptionReason
	if u.Reason != nil {
		reason = *u.Reason
	}
	if _, err := g.db.ExecContext(r.Context(),
		`UPDATE nodes SET tolerance_flag = ?, exception_reason = ? WHERE node_id = ?`,
		tolerance, nullFloat64(reason), nodeID); err != nil {
		g.serverError(w, err)
		return
	}

	if u.Requeue != nil && *u.Requeue {
		newID := nodeID + "-RQ"
		if _, err := g.db.ExecContext(r.Context(),
			`INSERT INTO nodes (node_id, path_id, label, distance, visit_time, tolerance_flag)
			 VALUES (?, ?, ?, ?, ?, 0)`,
			newID, nd.PathID, nd.Label, nd.Distance, nd.VisitTime, tolerance); err != nil {
			g.serverError(w, err)
			return
		}
	}

	updated, err := g.fetchNode(r.Context(), nodeID)
	if err != nil {
		g.serverError(w, err)
		return
	}
	writeJSON(w, http.StatusOK, updated)
}
